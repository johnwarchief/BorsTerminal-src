# Telegram Indicator Catalog (Stage A) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 64 photos and captions from the Telegram group `indicators` become real files inside `frontend-v2`, shown as a searchable 64-row reference list in the existing "اندیکاتورها و کانال" drawer tab.

**Architecture:** One Python tool turns a Telegram Desktop chat export into 64 WebP files plus a JSON catalog keyed `p01`…`p64`. A TypeScript module validates that JSON with zod at load time, resolves photo names through Vite's asset glob, and joins 9 rows to already-implemented overlays through a hand-maintained link table. One React component renders the list with its own scroll box and a lightbox; it is mounted as a third block inside the drawer's existing indicators panel. No backend, no API, no `fetch`.

**Tech Stack:** Python 3.14 stdlib (`html.parser`, `pathlib`) + Pillow 12.1.0 (WebP verified available) + pytest. React 18 + TypeScript strict + Vite 5 + Tailwind tokens + zod 4 + Vitest/Testing Library.

**Spec:** `docs/superpowers/specs/2026-11-03-telegram-indicator-catalog-design.md` — read it with this plan; the spec is the authority, this plan is the sequence.

**Branch:** `mission/4-axes-2026-10-10`. Spec commits already on it: `392204da`, `0243afd1`, `111c9518`.

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
tools/build_telegram_catalog_assets.py — از export تلگرام به assets برنامۀ مهاجرتی

ورودی: پوشۀ «Export chat history» گروه indicators (فقط Photos تیک خورده باشد)
و telegram_indicators_catalog.json که پیش‌تر ساخته شده.

خروجی: 64 فایل webp و telegram-indicators.json در frontend-v2.

قاعده: هر اختلافِ شمارش یا ابعاد بلند شکست می‌خورد، نه اینکه بی‌صدا یک
سطر بی‌عکس بسازد. همان باگی که در frontend/src/features/technical/lib/
tvIndicatorCatalog.ts:1 سِند شده است.
"""

from __future__ import annotations

import argparse
import json
import re
import sys
from dataclasses import dataclass, field
from html.parser import HTMLParser
from pathlib import Path
from typing import Any

MAX_WIDTH = 900
WEBP_QUALITY = 78
TOTAL_ROWS = 64
# تطبیقِ موقعیتیِ عکسِ export ← ردیفِ کاتالوگ. ابعادِ یکسان دو پستِ جابه‌جا را
# نمی‌گیرد، پس سطرِ اولِ کپشن سنسورِ مستقل است. تلگرام کپشن را در HTML کوتاه
# می‌کند، بنابراین تلورانس صفر نیست و لیستِ نخوانده‌ها چاپ می‌شود.
CAPTION_MISMATCH_TOLERANCE = 8


class CatalogMismatch(RuntimeError):
    """شمارش یا ابعادِ export با کاتالوگ نمی‌خواند. هیچ فایلی نوشته نمی‌شود."""


@dataclass
class PhotoEntry:
    src: str
    width: int | None
    height: int | None
    caption: str = ""

    @property
    def dims(self) -> str:
        if self.width and self.height:
            return f"{self.width}×{self.height}"
        return ""


@dataclass
class MatchedRow:
    id: str
    title_en: str
    caption_en: str
    photo: str
    dims: str
    attachments: list[str] = field(default_factory=list)


class _ExportParser(HTMLParser):
    """متن پیام و بعد هر img را بردارید. تلگرام کپشن را پیش از تصویر می‌گذارد."""

    def __init__(self) -> None:
        super().__init__(convert_charrefs=True)
        self.photos: list[PhotoEntry] = []
        self._in_text = 0
        self._text_parts: list[str] = []
        self._pending_caption = ""

    def handle_starttag(self, tag: str, attrs: list[tuple[str, str | None]]) -> None:
        a = dict(attrs)
        cls = a.get("class", "") or ""
        if tag == "div" and "message" in cls.split():
            self._text_parts = []
            self._pending_caption = ""
        if tag == "div" and "text_content" in cls.split():
            self._in_text += 1
        if tag == "br" and self._in_text:
            self._text_parts.append("\n")
        if tag == "img":
            w = _as_int(a.get("width"))
            h = _as_int(a.get("height"))
            if not self._pending_caption and self._text_parts:
                self._pending_caption = _clean("".join(self._text_parts))
            self.photos.append(
                PhotoEntry(
                    src=(a.get("src") or "").strip(),
                    width=w,
                    height=h,
                    caption=self._pending_caption,
                )
            )
            self._pending_caption = ""

    def handle_endtag(self, tag: str) -> None:
        if tag == "div" and self._in_text:
            self._in_text -= 1

    def handle_data(self, data: str) -> None:
        if self._in_text:
            self._text_parts.append(data)


def _as_int(value: str | None) -> int | None:
    try:
        return int(str(value).strip())
    except (TypeError, ValueError):
        return None


def _clean(text: str) -> str:
    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]
    return "\n".join(lines)


def _norm(text: str) -> str:
    return " ".join((text or "").lower().split())


def _first_line(text: str) -> str:
    for line in (text or "").splitlines():
        if line.strip():
            return line.strip()
    return ""


def caption_disagreements(
    catalog: list[dict[str, Any]], photos: list[PhotoEntry]
) -> list[str]:
    """ردیف‌هایی که سطرِ اولِ کپشنِ export با سطرِ اولِ عنوانِ کاتالوگ نمی‌خواند.
    بدونِ نرمال‌سازیِ واژگان: هر اختلافِ واقعی باید در لیستِ چاپ‌شده دیده شود."""
    if len(catalog) != len(photos):
        raise CatalogMismatch(
            f"caption cross-check needs equal lengths, got {len(catalog)} vs {len(photos)}"
        )
    bad: list[str] = []
    for i, (row, photo) in enumerate(zip(catalog, photos), start=1):
        html_cap = _norm(_first_line(photo.caption))
        cat_head = _norm(_first_line(row.get("title") or ""))
        if not html_cap or not cat_head:
            continue
        if html_cap[:40] != cat_head[:40]:
            bad.append(f"p{i:02d}")
    return bad


def parse_export(html_text: str) -> list[PhotoEntry]:
    p = _ExportParser()
    p.feed(html_text)
    return [ph for ph in p.photos if ph.src.lower().endswith((".jpg", ".jpeg", ".png", ".webp"))]


def match_rows(catalog: list[dict[str, Any]], photos: list[PhotoEntry]) -> list[MatchedRow]:
    if len(catalog) != len(photos):
        raise CatalogMismatch(
            f"catalog has {len(catalog)} rows but export has {len(photos)} photos"
        )
    out: list[MatchedRow] = []
    for i, (row, photo) in enumerate(zip(catalog, photos), start=1):
        cat_dims = (row.get("dimensions") or "").strip()
        if cat_dims and photo.dims and cat_dims != photo.dims:
            raise CatalogMismatch(
                f"row {i}: catalog says {cat_dims} but photo {photo.src} is {photo.dims}"
            )
        title = (row.get("title") or "").strip()
        lines = [ln.strip() for ln in title.splitlines() if ln.strip()]
        title_en = lines[0] if lines else ""
        caption_en = "\n".join(lines[1:])
        if title_en.lower() == "photo without caption":
            caption_en = ""
            title_en = ""
        out.append(
            MatchedRow(
                id=f"p{i:02d}",
                title_en=title_en,
                caption_en=caption_en,
                photo=f"p{i:02d}.webp",
                dims=cat_dims or photo.dims,
                attachments=[f["filename"] for f in row.get("files", []) if f.get("filename")],
            )
        )
    return out


def to_payload(rows: list[MatchedRow]) -> list[dict[str, Any]]:
    """نامِ فیلدها عمداً camelCase نوشته می‌شود، نه از __dict__ِ dataclass:
    کلیدهای JSON باید با schemaِ zod در telegramIndicatorCatalog.ts یکی باشند
    و `m.__dict__` آن‌ها را title_en/caption_en می‌داد."""
    return [
        {
            "id": r.id,
            "titleEn": r.title_en,
            "captionEn": r.caption_en,
            "photo": r.photo,
            "dims": r.dims,
            "attachments": list(r.attachments),
        }
        for r in rows
    ]


def convert_photos(photos: list[PhotoEntry], export_dir: Path, matches: list[MatchedRow], out_dir: Path) -> int:
    from PIL import Image

    out_dir.mkdir(parents=True, exist_ok=True)
    total = 0
    for photo, row in zip(photos, matches):
        src = (export_dir / photo.src).resolve()
        if not src.is_file():
            raise CatalogMismatch(f"{row.id}: source photo missing at {src}")
        with Image.open(src) as im:
            im = im.convert("RGB")
            if im.width > MAX_WIDTH:
                im = im.resize((MAX_WIDTH, round(im.height * MAX_WIDTH / im.width)))
            dst = out_dir / row.photo
            im.save(dst, "WEBP", quality=WEBP_QUALITY, method=6)
            total += dst.stat().st_size
    return total


def main(argv: list[str] | None = None) -> int:
    if hasattr(sys.stdout, "reconfigure"):
        sys.stdout.reconfigure(encoding="utf-8")
    ap = argparse.ArgumentParser(description=__doc__)
    ap.add_argument("--export-dir", required=True, type=Path)
    ap.add_argument("--catalog", default=Path("telegram_indicators_catalog.json"), type=Path)
    ap.add_argument(
        "--out-dir",
        default=Path("frontend-v2/src/features/technical/assets/telegram-indicators"),
        type=Path,
    )
    ap.add_argument(
        "--json-out",
        default=Path("frontend-v2/src/features/technical/data/telegram-indicators.json"),
        type=Path,
    )
    ap.add_argument("--dry-run", action="store_true", help="فقط تطبیق و گزارش؛ چیزی ننویس")
    args = ap.parse_args(argv)

    html_candidates = sorted(args.export_dir.glob("**/history.html"))
    if not html_candidates:
        raise CatalogMismatch(f"no history.html under {args.export_dir}")
    html_text = html_candidates[0].read_text(encoding="utf-8", errors="replace")

    catalog = json.loads(args.catalog.read_text(encoding="utf-8"))
    photos = parse_export(html_text)
    matches = match_rows(catalog, photos)
    if len(matches) != TOTAL_ROWS:
        raise CatalogMismatch(f"expected {TOTAL_ROWS} rows, matched {len(matches)}")

    disagreements = caption_disagreements(catalog, photos)
    no_title = [m.id for m in matches if not m.title_en]
    no_caption = [m.id for m in matches if not m.caption_en]
    print(f"rows: {len(matches)}")
    print(f"rows with no title: {len(no_title)} {no_title}")
    print(f"rows with no caption body: {len(no_caption)}")
    print(
        f"caption cross-check: {len(disagreements)} of {len(matches)} disagree "
        f"(tolerance {CAPTION_MISMATCH_TOLERANCE}) {disagreements}"
    )
    if len(disagreements) > CAPTION_MISMATCH_TOLERANCE:
        raise CatalogMismatch(
            "export order does not match the catalog order on "
            f"{len(disagreements)} rows — refusing to pair positionally"
        )

    if args.dry_run:
        print("dry-run: nothing written")
        return 0

    root = html_candidates[0].parent
    total_bytes = convert_photos(photos, root, matches, args.out_dir)
    args.json_out.parent.mkdir(parents=True, exist_ok=True)
    args.json_out.write_text(
        json.dumps(to_payload(matches), ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
        newline="\n",
    )
    print(f"wrote {len(matches)} webp files, {total_bytes / 1_048_576:.2f} MB total")
    print(f"wrote {args.json_out}")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())

  id: z.string().regex(/^p\d{2}$/),
  titleEn: z.string(),
  captionEn: z.string(),
  photo: z.string().regex(/^p\d{2}\.webp$/),
  dims: z.string(),
  attachments: z.array(z.string()),

export type TelegramRow = z.infer<typeof RowSchema> & {
  photoUrl: string;
  overlayId: string | null;
};

const photoModules = import.meta.glob('../assets/telegram-indicators/*.webp', {
  eager: true,
  query: '?url',
  import: 'default',
}) as Record<string, string>;

  Object.entries(photoModules).map(([path, url]) => [path.split('/').pop() as string, url]),
);

/**
 * ردیف‌های کاتالوگ ← اورلیِ پیاده‌شده در همین درخت. دستی نگه داشته می‌شود و
 * در JSONِ تولیدی نیست تا با تولیدِ دوبله گم نشود.
 * p25 و p40 تکراریِ همان پست‌های p19 و p11 هستند، پس لینک می‌گیرند.
 */
export const TELEGRAM_OVERLAY_LINKS: Record<string, string> = {
  p02: 'FiboRSI8',
  p03: 'SuperTrendHeikinAshi',
  p05: 'RsiMaSmooth',
  p06: 'BreakoutRetestPro',
  p09: 'DoubleTap3D',
  p11: 'AutoTrendlineCHoCH',
  p19: 'SuperTrendPullback',
  p25: 'SuperTrendPullback',
  p40: 'AutoTrendlineCHoCH',
};

/** دو اورلیِ پیاده‌شده که هیچ پستِ تلگرامی بی‌ابهام نامشان نمی‌برد — حدس زده نشد */
export const UNLINKED_OVERLAYS: string[] = ['MultiModeChannel', 'MultiVariantRSI'];

function assertCatalog(rows: ReturnType<typeof CatalogSchema.parse>): void {
  if (rows.length !== 64) {
    throw new Error(`telegram catalog: expected 64 rows, got ${rows.length}`);
  }
  const noPhoto = rows.filter((r) => !urlByName.has(r.photo)).map((r) => r.id);
  if (noPhoto.length > 0) {
    throw new Error(`telegram catalog: photos missing from assets for ${noPhoto.join(', ')}`);
  }
  const known = new Set([
    ...ADVANCED_INDICATORS_REGISTRY.map((i) => i.id),
    ...USER_INDICATORS_REGISTRY.map((i) => i.id),
  ]);
  const dead = Object.entries(TELEGRAM_OVERLAY_LINKS).filter(([, id]) => !known.has(id));
  if (dead.length > 0) {
    throw new Error(`telegram catalog: links to absent overlays: ${JSON.stringify(dead)}`);

  photoUrl: urlByName.get(r.photo) as string,
  overlayId: TELEGRAM_OVERLAY_LINKS[r.id] ?? null,
}));

      r.titleEn.toLowerCase().includes(q) ||
      r.captionEn.toLowerCase().includes(q) ||
      r.attachments.some((a) => a.toLowerCase().includes(q)),
  );
}

  const [openRow, setOpenRow] = useState<TelegramRow | null>(null);
  const lightboxRef = useRef<HTMLDivElement | null>(null);

  const active = useTechnicalStore((s) => s.activeUserIndicators);
  const toggle = useTechnicalStore((s) => s.toggleUserIndicator);

  const rows = useMemo(() => searchTelegramRows(query), [query]);

  const closeLightbox = useCallback(() => setOpenRow(null), []);

  useEffect(() => {
    if (!openRow) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === 'Escape') setOpenRow(null);
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [openRow]);

  return (
    <div className={`flex flex-col gap-2 ${className}`} data-testid="telegram-catalog-list">
      <div className="flex items-center justify-between gap-2">
        <span className="t-10 font-bold text-accent">کاتالوگ تلگرام ({toFaDigits(TELEGRAM_CATALOG.length)})</span>
        <span className="t-85 text-text-3 font-mono" data-testid="telegram-catalog-count">
          {toFaDigits(rows.length)} سطر
        </span>
      </div>

      <input
        type="search"
        data-testid="telegram-catalog-search"
        value={query}
        onChange={(e) => setQuery(e.target.value)}
        placeholder="جستجو در نام، توضیح یا فایل"
        aria-label="جستجو در کاتالوگ تلگرام"
        className="h-7 w-full rounded-[var(--radius-xs)] border border-line bg-panel px-2 t-9 text-text placeholder:text-text-3 focus:outline-none focus:border-accent"
      />

      <div
        className="overflow-y-auto rounded-[var(--radius-sm)] border border-line"
        style={{ maxHeight: `${MAX_HEIGHT_PX}px` }}
      >
        {rows.map((r) => {
          const overlayOn = r.overlayId != null && active.includes(r.overlayId);
          return (
            <div
              key={r.id}
              data-testid={`telegram-row-${r.id}`}
              className="flex items-center gap-2 border-b border-line/40 px-2 py-1 last:border-b-0 odd:bg-panel-2/30"
              style={{ minHeight: '52px' }}
            >
              <button
                type="button"
                data-testid={`telegram-thumb-${r.id}`}
                onClick={() => setOpenRow(r)}
                aria-label={`نمایش تصویر ${r.titleEn || r.id}`}
                className="shrink-0 overflow-hidden rounded-[var(--radius-xs)] border border-line bg-panel"
                style={{ width: `${THUMB_W}px`, height: `${THUMB_H}px` }}
              >
                <img
                  src={r.photoUrl}
                  alt=""
                  width={THUMB_W}
                  height={THUMB_H}
                  loading="lazy"
                  decoding="async"
                  className="h-full w-full object-cover"
                />
              </button>

              <div className="flex min-w-0 flex-1 flex-col gap-0.5">
                <span className="t-95 truncate font-bold text-text">{r.titleEn || r.attachments[0] || r.id}</span>
                <span
                  data-testid={`telegram-caption-${r.id}`}
                  className="t-8 line-clamp-1 text-text-3"
                >
                  {r.captionEn || 'بدون کپشن'}
                </span>
              </div>

              <span className="t-8 shrink-0 font-mono text-text-3">{r.dims || '—'}</span>

              <div className="shrink-0" data-testid={`telegram-status-${r.id}`}>
                {r.overlayId == null ? (
                  <span className="t-8 text-text-3">پیاده‌سازی نشده</span>
                ) : (
                  <button
                    type="button"
                    data-testid={`toggle-indicator-${r.id}`}
                    onClick={() => toggle(r.overlayId as string)}
                    className={`flex items-center gap-1 rounded-[var(--radius-xs)] border px-2 t-9 font-medium transition-colors ${
                      overlayOn
                        ? 'border-emerald-500/40 bg-emerald-500/20 font-bold text-emerald-400'
                        : 'border-line bg-panel text-text-3 hover:text-text'
                    }`}
                  >
                    <span>{overlayOn ? 'روشن' : 'خاموش'}</span>
                    <span className={`h-1.5 w-1.5 rounded-full ${overlayOn ? 'bg-emerald-400' : 'bg-gray-500'}`} />
                  </button>
                )}
              </div>
            </div>
          );
        })}

        {rows.length === 0 && (
          <div className="p-4 t-9 text-center text-text-3">موردی با این عبارت یافت نشد.</div>
        )}
      </div>

      {openRow && (
        <div
          ref={lightboxRef}
          data-testid="telegram-lightbox"
          role="dialog"
          aria-modal="true"
          aria-label={openRow.titleEn || openRow.id}
          tabIndex={-1}
          onClick={closeLightbox}
          className="fixed inset-0 z-50 flex items-center justify-center bg-black/70 p-4 telegram-catalog-lightbox"
        >
          <div
            className="flex max-h-full max-w-4xl flex-col gap-2 rounded-[var(--radius-md)] border border-line bg-panel p-3"
            onClick={(e) => e.stopPropagation()}
          >
            <img
              src={openRow.photoUrl}
              alt={openRow.titleEn || openRow.id}
              className="max-h-[70vh] w-auto rounded-[var(--radius-sm)] object-contain"
            />
            <div className="flex flex-col gap-1">
              <span className="t-10 font-bold text-text">{openRow.titleEn || openRow.attachments[0] || openRow.id}</span>
              {openRow.captionEn && (
                <p className="t-85 whitespace-pre-line leading-snug text-text-2">{openRow.captionEn}</p>
              )}
              {openRow.attachments.length > 0 && (
                <span className="t-8 font-mono text-text-3">{openRow.attachments.join('، ')}</span>
              )}
            </div>
          </div>
        </div>
      )}
    </div>
  );
}
```
