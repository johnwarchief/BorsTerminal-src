// ---------------------------------------------------------------- comparisons
// Pine treats floats closer than about 1e-10 as equal in comparisons (measured on the stage-2 ES 1m export,
// 2026-10-30). The Python reference compares exactly, so every float comparison here goes through these helpers:
// the difference is scaled far above the tolerance and compared with zero, which is exact for the values used.
float CMP = 1e20
gtE(float a, float b) => (a - b) * CMP > 0
geE(float a, float b) => (a - b) * CMP >= 0
ltE(float a, float b) => (b - a) * CMP > 0
leE(float a, float b) => (b - a) * CMP >= 0
eqE(float a, float b) => (a - b) * CMP == 0
maxE(float a, float b) => ltE(a, b) ? b : a
minE(float a, float b) => gtE(a, b) ? b : a
absE(float a) => ltE(a, 0.0) ? -a : a

// ---------------------------------------------------------------- engine (spec v0.2 2.1-2.4)
type Engine
    float k
    int   n
    bool  closes  = false
    int   nTr     = 0
    float trSum   = 0.0
    float rma     = na        // RMA(TR, n) before the floor
    float atr     = na        // after the floor
    float cPrev   = na
    bool  started = false
    int   sid     = 0
    float dir     = 1.0
    int   b0      = na
    float p0      = na
    float atr0    = na
    float pe      = na
    int   be      = na
    float atre    = na
    float deep    = na
    int   birth   = na
    bool  conf    = false     // a swing was confirmed on this bar
    int   cB0     = na        // the confirmed swing: pivots, direction, A, D
    int   cB1     = na
    float cP0     = na
    float cP1     = na
    float cDir    = na
    float cA      = na
    int   cD      = na

// One confirmed bar. t = bars since the start; floorV = the ATR floor on this bar (2 mintick).
method step(Engine e, int t, float h, float l, float c, float floorV) =>
    // 2.4: TR on the true range in both modes (A14); RMA seeded with the SMA of the first n TRs; floor after the RMA
    float tr = na(e.cPrev) ? h - l : maxE(maxE(h - l, absE(h - e.cPrev)), absE(l - e.cPrev))
    e.nTr := e.nTr + 1
    if e.nTr < e.n
        e.trSum := e.trSum + tr
    else if e.nTr == e.n
        e.rma := (e.trSum + tr) / e.n
    else
        e.rma := (e.rma * (e.n - 1) + tr) / e.n
    e.atr := e.nTr >= e.n ? maxE(e.rma, floorV) : na
    e.cPrev := c
    float hh = e.closes ? c : h
    float ll = e.closes ? c : l
    e.conf := false
    if not e.started
        // 2.3: provisional up swing on the first bar with a defined ATR
        if not na(e.atr)
            e.started := true
            e.dir   := 1.0
            e.b0    := t
            e.p0    := ll
            e.atr0  := e.atr
            e.pe    := hh
            e.be    := t
            e.atre  := e.atr
            e.deep  := c
            e.birth := t
    else
        // 2.2, written as an up swing on mirrored prices
        float d     = e.dir
        float hi    = d > 0 ? hh : -ll
        float lo    = d > 0 ? ll : -hh
        float cl    = d > 0 ? c : -c
        float peU   = e.pe * d
        float deepU = e.deep * d
        if gtE(hi, peU)                   // step 1, extension
            peU    := hi
            e.be   := t
            e.atre := e.atr
            deepU  := cl
        else                              // step 2
            deepU := minE(deepU, lo)
        e.pe   := peU * d
        e.deep := deepU * d
        if geE(peU - deepU, e.k * e.atre) // step 3, at most one confirmation per bar
            e.conf  := true
            e.cB0   := e.b0
            e.cB1   := e.be
            e.cP0   := e.p0
            e.cP1   := e.pe
            e.cDir  := d
            e.cA    := d * (e.pe - e.p0) / e.atr0
            e.cD    := e.be - e.b0
            e.sid   := e.sid + 1
            e.dir   := -d
            e.p0    := e.pe
            e.b0    := e.be
            e.atr0  := e.atre
            e.pe    := e.deep
            e.be    := t
            e.atre  := e.atr
            e.deep  := c
            e.birth := t
    e

// 2.1 state quantities of the live swing on the bar of close c
method rhoC(Engine e, float c) =>
    e.dir * (e.pe - c) / (e.k * e.atre)
method extent(Engine e) =>
    e.dir * (e.pe - e.p0) / e.atr0

// ---------------------------------------------------------------- tables and records (v0.2 2.6, 4.1; v0.3 B2; v0.3.2 S1)
// Flat cell index dir*80 + rho*16 + g*4 + r (spec 9); dir index 0 = up, 1 = down. Returns [cell, clipped rho_c, rho bin].
cellOf(float d, float rhoC, int g, float r) =>
    float rc = minE(maxE(rhoC, 0.0), 1.0)
    int ri = math.min(int(rc / 0.2), 4)
    int gi = g <= 0 ? 0 : g <= 2 ? 1 : g <= 5 ? 2 : 3
    int xi = ltE(r, 1.5) ? 0 : ltE(r, 2.5) ? 1 : ltE(r, 4.0) ? 2 : 3
    int di = d > 0 ? 0 : 1
    [di * 80 + ri * 16 + gi * 4 + xi, rc, ri]

type Book
    array<float> hz          // 160 cells x 5: n, sum Y_in, sum Y_N, sum Y_C, sum (Y_in - rho_c)
    array<float> srt0        // sorted A of kept swings, up / down
    array<float> srt1
    array<float> ringA0      // ring buffers of A and D, N_MAX per direction
    array<float> ringA1
    array<int>   ringD0
    array<int>   ringD1
    array<int>   liveT       // the live swing's bars so far: bar, cell, clipped rho_c
    array<int>   liveC
    array<float> liveR
    int pos0  = 0
    int pos1  = 0
    int kept0 = 0
    int kept1 = 0

newBook() =>
    Book.new(array.new<float>(800, 0.0), array.new<float>(), array.new<float>(), array.new<float>(), array.new<float>(),
         array.new<int>(), array.new<int>(), array.new<int>(), array.new<int>(), array.new<float>())

addAt(array<float> a, int i, float v) =>
    array.set(a, i, array.get(a, i) + v)

// number of stored values <= x, and < x (sorted array)
countLE(array<float> s, float x) =>
    int lo = 0
    int hi = array.size(s)
    while lo < hi
        int mid = int((lo + hi) / 2)
        if leE(array.get(s, mid), x)
            lo := mid + 1
        else
            hi := mid
    lo
countLT(array<float> s, float x) =>
    int lo = 0
    int hi = array.size(s)
    while lo < hi
        int mid = int((lo + hi) / 2)
        if ltE(array.get(s, mid), x)
            lo := mid + 1
        else
            hi := mid
    lo
countX(array<float> s, float x, bool left) =>
    int n = 0
    if left
        n := countLT(s, x)
    else
        n := countLE(s, x)
    n

method insertRec(Book bk, int di, float A, int D) =>
    array<float> srt = di == 0 ? bk.srt0 : bk.srt1
    array<float> ra  = di == 0 ? bk.ringA0 : bk.ringA1
    array<int>   rd  = di == 0 ? bk.ringD0 : bk.ringD1
    int pos = di == 0 ? bk.pos0 : bk.pos1
    if array.size(ra) == N_MAX
        float old = array.get(ra, pos)
        array.remove(srt, countLT(srt, old))
        array.set(ra, pos, A)
        array.set(rd, pos, D)
    else
        array.push(ra, A)
        array.push(rd, D)
    if di == 0
        bk.pos0 := (pos + 1) % N_MAX
    else
        bk.pos1 := (pos + 1) % N_MAX
    array.insert(srt, countLE(srt, A), A)

// Coarser tables the cells shrink toward (they replace the twin as the prior):
var array<float> margR = array.new<float>(40, 0.0)     // rho_c margin [dir, rho bin]: n, sum Y_N, sum Y_C, sum (Y_in - rho_c)
var array<float> baseR = array.new<float>(6, 0.0)      // direction base rate [dir]: n, sum Y_N, sum Y_C

// After e.step() on bar t: a swing confirmed on t enters the tables and records (spec 5: before the forecast of t),
// then bar t joins the live swing's list. Each labelled bar adds 1 to its cell's n and Y to its sums (S1).
method absorb(Book bk, Engine e, int t, float c) =>
    if e.started
        if e.conf
            int n = array.size(bk.liveT)
            if e.sid - 1 >= DISCARD and n > 0
                int di = e.cDir > 0 ? 0 : 1
                if di == 0
                    bk.kept0 := bk.kept0 + 1
                else
                    bk.kept1 := bk.kept1 + 1
                for i = 0 to n - 1
                    int tt   = array.get(bk.liveT, i)
                    int cc   = array.get(bk.liveC, i)
                    float rv = array.get(bk.liveR, i)
                    float yin = e.cB1 <= tt ? 1.0 : 0.0
                    float yn  = e.cB1 <= tt + HZN ? 1.0 : 0.0
                    float yc  = t <= tt + HZN ? 1.0 : 0.0
                    addAt(bk.hz, cc * 5, 1.0)
                    addAt(bk.hz, cc * 5 + 1, yin)
                    addAt(bk.hz, cc * 5 + 2, yn)
                    addAt(bk.hz, cc * 5 + 3, yc)
                    addAt(bk.hz, cc * 5 + 4, yin - rv)
                    int rb = math.min(int(rv / 0.2), 4)
                    int mi = (di * 5 + rb) * 4
                    addAt(margR, mi, 1.0)
                    addAt(margR, mi + 1, yn)
                    addAt(margR, mi + 2, yc)
                    addAt(margR, mi + 3, yin - rv)
                    addAt(baseR, di * 3, 1.0)
                    addAt(baseR, di * 3 + 1, yn)
                    addAt(baseR, di * 3 + 2, yc)
                bk.insertRec(di, e.cA, e.cD)
            array.clear(bk.liveT)
            array.clear(bk.liveC)
            array.clear(bk.liveR)
        [cell, rc, ri] = cellOf(e.dir, e.rhoC(c), t - e.be, e.extent() / e.k)
        array.push(bk.liveT, t)
        array.push(bk.liveC, cell)
        array.push(bk.liveR, rc)
    bk

// ---------------------------------------------------------------- band (v0.2 4.2; v0.3 B3; v0.3.1 R4, R9; v0.3.2 S3)
// {A > a} subset of the records: [pooled with the other direction, count base, n]
subset(Book bk, int di, float a) =>
    array<float> own = di == 0 ? bk.srt0 : bk.srt1
    array<float> oth = di == 0 ? bk.srt1 : bk.srt0
    int le    = countLE(own, a)
    int nOwn  = array.size(own) - le
    bool pooled = nOwn < N_POOL
    int base  = le
    int n     = nOwn
    if pooled
        int le2 = countLE(oth, a)
        base := le + le2
        n    := nOwn + array.size(oth) - le2
    [pooled, base, n]

// excess CDF G(x) (left: the left limit G(x-)): the empirical CDF of this chart's records blended toward Exp(mu)
gExcess(float x, float a, float mu, int di, Book bk, bool pr, int br, int nr, bool left) =>
    float g0 = 1.0 - math.exp(-x / mu)
    float g = g0
    if nr > 0
        float ax = a + x
        int cnt = countX(di == 0 ? bk.srt0 : bk.srt1, ax, left) - br
        if pr
            cnt := cnt + countX(di == 0 ? bk.srt1 : bk.srt0, ax, left)
        float w = nr / (nr + M_BAND)
        g := w * (float(cnt) / nr) + (1.0 - w) * g0
    g

gMix(float x, float p, float a, float mu, int di, Book bk, bool pr, int br, int nr) =>
    p + (1.0 - p) * gExcess(x, a, mu, di, bk, pr, br, nr, false)

// chance of a tag from here: the swing keeps going (1 - P_in) and its final extent reaches the level,
// P(A >= lv) = (1 - P_in) * (1 - G((lv - a)-)); 1 when the level is at or inside the extreme already
tagOdds(float lv, float a, float p, float mu, int di, Book bk, bool pr, int br, int nr) =>
    float out = na
    if not na(lv)
        out := leE(lv, a) ? 1.0 : (1.0 - p) * (1.0 - gExcess(lv - a, a, mu, di, bk, pr, br, nr, true))
    out

// R4 with step 0 (S3): x = 0 exactly when u <= P_in; otherwise bisection, warm-started from the previous bar's x (pine_mode)
solveX(float u, float p, float a, float mu, int di, Book bk, bool pr, int br, int nr, bool warm, float xPrev) =>
    float x = 0.0
    if gtE(u, p)
        float lo = 0.0
        float hi = 1.0
        int steps = FULL_STEPS
        if warm
            float d = DELTA
            bool found = false
            while not found
                lo := maxE(0.0, xPrev - d)
                hi := xPrev + d
                if leE(gMix(lo, p, a, mu, di, bk, pr, br, nr), u) and leE(u, gMix(hi, p, a, mu, di, bk, pr, br, nr))
                    found := true
                else
                    d := d * 2.0
            steps := WARM_STEPS
        else
            while ltE(gMix(hi, p, a, mu, di, bk, pr, br, nr), u)
                hi := hi * 2.0
        for i = 1 to steps
            float mid = 0.5 * (lo + hi)
            if geE(gMix(mid, p, a, mu, di, bk, pr, br, nr), u)
                hi := mid
            else
                lo := mid
        x := 0.5 * (lo + hi)
    x

// ---------------------------------------------------------------- window (v0.2 4.3; v0.3 A9), at each swing's birth
// D of the records with A > a and D > d: own direction, the other direction added while fewer than N_POOL
windowVals(Book bk, int di, float a, int d) =>
    array<float> v = array.new<float>()
    for side = 0 to 1
        int dd = side == 0 ? di : 1 - di
        if side == 1 and array.size(v) >= N_POOL
            break
        array<float> ra = dd == 0 ? bk.ringA0 : bk.ringA1
        array<int>   rd = dd == 0 ? bk.ringD0 : bk.ringD1
        int n = array.size(ra)
        if n > 0
            for i = 0 to n - 1
                if gtE(array.get(ra, i), a) and array.get(rd, i) > d
                    array.push(v, float(array.get(rd, i)))
    array.sort(v, order.ascending)
    v

// Quantile at level u of the mixture: the terminal is the extreme so far (d_ext) with probability p, otherwise
// drawn from the sorted values s; na when s is empty
windowQ(float u, float p, float dExt, array<float> s) =>
    float out = na
    if leE(u, p)
        out := dExt
    else
        int n = array.size(s)
        if n > 0
            float v = (u - p) / (1.0 - p)
            int i = 0
            while i < n and na(out)
                float xq = array.get(s, i)
                if geE(float(countLE(s, xq)) / n, v - 1e-12)
                    out := xq
                i := i + 1
    out

// ---------------------------------------------------------------- kernel trend (Nadaraya-Watson: rational quadratic + gaussian)
// Same maths as jdehorty/KernelFunctions, written inline so no library import is needed. Past bars only: no repaint on closed bars.
krRQ(float s, int lb, float rw, int st) =>
    float cw = 0.0
    float sw = 0.0
    for i = 0 to st + 1
        float w = math.pow(1.0 + (i * i) / (math.pow(lb, 2) * 2.0 * rw), -rw)
        cw += s[i] * w
        sw += w
    cw / sw
krGauss(float s, int lb, int st) =>
    float cw = 0.0
    float sw = 0.0
    for i = 0 to st + 1
        float w = math.exp(-(i * i) / (2.0 * math.pow(lb, 2)))
        cw += s[i] * w
        sw += w
    cw / sw
