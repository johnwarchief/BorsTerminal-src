f_tf() =>
    int m = timeframe.multiplier
    string r = str.tostring(m) + "M"
    if timeframe.isseconds
        r := str.tostring(m) + "s"
    else if timeframe.isticks
        r := str.tostring(m) + "t"
    else if timeframe.isminutes
        r := m % 60 == 0 ? str.tostring(math.floor(m / 60)) + "H" : str.tostring(m) + "m"
    else if timeframe.isdaily
        r := str.tostring(m) + "D"
    else if timeframe.isweekly
        r := str.tostring(m) + "W"
    r

f_abbr(float v) =>
    float a = math.abs(v)
    float r0 = math.round(a, 2)
    float rk = math.round(a / 1000, 1)
    float rm = math.round(a / 1000000, 1)
    float rb = math.round(a / 1000000000, 1)
    string body = r0 < 1000 ? str.tostring(r0, "0.##") : rk < 1000 ? str.tostring(rk, "0.#") + "K" : rm < 1000 ? str.tostring(rm, "0.#") + "M" : str.tostring(rb, "0.#") + "B"
    na(v) ? "—" : (v < 0 ? "-" : "") + body

f_esc(string s) =>
    str.replace_all(str.replace_all(s, "\\", "\\\\"), "\"", "\\\"")

f_pos(string p) =>
    switch p
        "Top Left" => position.top_left
        "Bottom Right" => position.bottom_right
        "Bottom Left" => position.bottom_left
        => position.top_right

f_q(float v) =>
    na(v) ? "null" : "\"" + str.tostring(v, format.mintick) + "\""

f_entryPayload(string actWord, string dir, float en, float sl, float t1, float t2, float t3) =>
    string inner = str.format('"action":"{0}","ticker":"{1}","tf":"{2}","direction":"{3}","entry":{4},"sl":{5},"tp1":{6},"tp2":{7},"tp3":{8}', f_esc(actWord), f_esc(syminfo.tickerid), f_esc(timeframe.period), dir, f_q(en), f_q(sl), f_q(t1), f_q(t2), f_q(t3))
    "{" + inner + "}"

f_closePayload(string actWord, string dir, float en, float px, float sl, string why) =>
    string inner = str.format('"action":"{0}","ticker":"{1}","tf":"{2}","direction":"{3}","entry":{4},"price":{5},"sl":{6},"reason":"{7}"', f_esc(actWord), f_esc(syminfo.tickerid), f_esc(timeframe.period), dir, f_q(en), f_q(px), f_q(sl), why)
    "{" + inner + "}"

f_pctTxt(float lvl, float entry, bool isLong) =>
    float p = (isLong ? lvl - entry : entry - lvl) / entry * 100
    (p >= 0 ? "+" : "") + str.tostring(p, "0.00") + "%"

f_del(BoxSet o) =>
    if not na(o.bx)
        box.delete(o.bx)
    if not na(o.upLn)
        line.delete(o.upLn)
    if not na(o.dnLn)
        line.delete(o.dnLn)
    if not na(o.mk)
        label.delete(o.mk)
    o

f_put(table t, int r, string lbl, string val, color vc, string tip) =>
    color bg = r % 2 == 1 ? C_DASH_BG : color.new(C_DASH_BG, 40)
    table.cell(t, 0, r, "  " + lbl, text_color = color.new(C_DASH_TXT, 25), text_halign = text.align_left, text_size = size.small, bgcolor = bg, tooltip = tip)
    table.cell(t, 1, r, val + "  ", text_color = vc, text_halign = text.align_right, text_size = size.small, bgcolor = bg)
    r + 1

f_warn(table t, int r, string lbl, string val, string tip) =>
    table.cell(t, 0, r, "  " + lbl, text_color = color.new(C_DASH_TXT, 25), text_halign = text.align_left, text_size = size.small, bgcolor = C_WARN, tooltip = tip)
    table.cell(t, 1, r, val + "  ", text_color = C_DASH_TXT, text_halign = text.align_right, text_size = size.small, bgcolor = C_WARN)
    r + 1

f_smooth(float src, string m, int len) =>
    float vS = ta.sma(src, len)
    float vE = ta.ema(src, len)
    float vW = ta.wma(src, len)
    float vR = ta.rma(src, len)
    float vH = ta.hma(src, len)
    switch m
        "SMA" => vS
        "EMA" => vE
        "WMA" => vW
        "RMA" => vR
        "HMA" => vH
        => src

f_htf(int pBox, int pAtr, float pThr, float pMult, int pNorm, string pSm, int pSmLen) =>
    float xAtr = f_smooth(ta.atr(pAtr), pSm, pSmLen)
    float xHi = ta.highest(high, pBox)
    float xLo = ta.lowest(low, pBox)
    float xMed = ta.median(xAtr, pNorm)
    int xLeft = time[pBox - 1]
    var int xCnt = 0
    if not na(xAtr)
        xCnt += 1
    float xNorm = xCnt >= pNorm ? xMed : na
    bool xQuiet = bar_index >= pBox - 1 and not na(xHi) and not na(xLo) and not na(xAtr) and not na(xLeft) and (xHi - xLo) < pThr * xAtr
    var bool xOn = false
    var float xUl = float(na)
    var float xDl = float(na)
    int xCd = 0
    if xOn
        bool xU = close > xUl
        bool xD = xDl > 0 and close < xDl
        if xU or xD
            xOn := false
            xCd := xU ? 1 : 2
    else if xQuiet and not na(xNorm)
        xOn := true
        xUl := math.round_to_mintick(xHi + pMult * xNorm)
        xDl := math.round_to_mintick(xLo - pMult * xNorm)
    xCd[1]

    t1Lb := label.new(bar_index, nT1, "◆ TP1 " + str.tostring(nT1, format.mintick), style = label.style_label_left, color = color.new(C_TP, 0), textcolor = C_MTX, size = size.small)
    t2Lb := not na(nT2) ? label.new(bar_index, nT2, "✦ TP2 " + str.tostring(nT2, format.mintick), style = label.style_label_left, color = color.new(C_TP, 0), textcolor = C_MTX, size = size.small) : label(na)
    t3Lb := not na(nT3) ? label.new(bar_index, nT3, "◆ TP3 " + str.tostring(nT3, format.mintick), style = label.style_label_left, color = color.new(C_TP, 0), textcolor = C_MTX, size = size.small) : label(na)
