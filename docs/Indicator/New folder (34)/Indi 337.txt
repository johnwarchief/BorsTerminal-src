// Retracement ratios (0 … 1.0 only) and continuation projections — two
// SEPARATE concepts with separate price functions (no mixed coordinate
// system, no discontinuity at 1.0).
var retraceRatios = array.from(0.0, 0.236, 0.382, 0.5, 0.618, 0.786, 1.0)
var projMults     = array.from(1.272, 1.618)

// Retracement price: 0.0 anchored at the impulse END, 1.0 at the START, so
// 0.5–0.618 sits deep in the pullback (bullish → below price as support,
// bearish → above price as resistance).
f_retracement_price(bull, hi, lo, r) =>
    rng = hi - lo
    bull ? hi - r * rng : lo + r * rng

// Projection price: continuation target beyond the impulse END, as a multiple
// of the impulse range (1.272× → 0.272 × range beyond the extreme).
f_projection_price(bull, hi, lo, m) =>
    rng = hi - lo
    bull ? hi + (m - 1.0) * rng : lo - (m - 1.0) * rng

// ─────────────────────────────────────────────────────────────────────────────
// 3. SWING DETECTION (confirmed pivots, processed on CLOSED bars only)
// ─────────────────────────────────────────────────────────────────────────────
// A pivot value is non-na only on its confirmation bar (`confirmBars` after the
// actual turning point) and is consumed only when that bar CLOSES.
phLocal = ta.pivothigh(pivotLen, confirmBars)
plLocal = ta.pivotlow(pivotLen, confirmBars)
newPH = barstate.isconfirmed and not na(phLocal) ? phLocal : na
newPL = barstate.isconfirmed and not na(plLocal) ? plLocal : na

// ── ZigZag state ──
atrZZ = ta.atr(14)
var float zzP0 = na     // previous confirmed ZigZag point (price) — the leg's far end
var int   zzX0 = na
var float zzP1 = na     // latest confirmed ZigZag point (price)
var int   zzX1 = na
var int   zzD1 = 0      // +1 = latest point is a high, -1 = a low, 0 = none yet
var line  zzSeg = na    // the latest segment (zzP0 → zzP1), extended on continuation
var label zzLbl = na    // label on the latest point; moved/relabeled while it forms
var float zzHigh     = na   // latest ZigZag high  + the one before it
var float zzPrevHigh = na
var float zzLow      = na   // latest ZigZag low   + the one before it
var float zzPrevLow  = na
var string bias = "Neutral"
zzLegEvent = false      // true on the bar the latest leg changes (flip or extend)

// Leg-size filter (repair #7):
//  • zzDevAtr = 0 → filter FULLY disabled (threshold 0 from bar 1).
//  • zzDevAtr > 0 → threshold = zzDevAtr × ATR at the CANDIDATE PIVOT bar
//    (atrZZ[confirmBars]), matching the swing being measured. If that ATR is
//    still na (warm-up), the >= comparison below evaluates na → the reversal
//    is REJECTED — the filter is never silently bypassed.
pivotAtr = atrZZ[confirmBars]
zzMinLeg = zzDevAtr == 0.0 ? 0.0 : zzDevAtr * pivotAtr

// ── Simultaneous pivot high + low on one candle (repair #6) ──
// A large outside candle can confirm BOTH a pivot high and a pivot low on the
// same bar. Processing both would allow two state transitions per candle or a
// zero-duration vertical leg. Only ONE pivot is processed per bar: the
// candidate farther from the current endpoint wins; on a tie, or when no
// endpoint exists yet, the ambiguous pair is discarded entirely.
// (Same-bar-index anchors are impossible after this: any two pivots sharing a
// bar arrive on the same confirmation bar and are resolved here, and older
// endpoints always sit on strictly earlier bars.)
float usePH = newPH
float usePL = newPL
if not na(newPH) and not na(newPL)
    if na(zzP1)
        usePH := na
        usePL := na
    else
        dH = math.abs(newPH - zzP1)
        dL = math.abs(newPL - zzP1)
        if dH > dL
            usePL := na
        else if dL > dH
            usePH := na
        else
            usePH := na
            usePL := na

// Confirmed-pivot bar indices (the ZigZag reads the pivot PRICE from usePH/usePL
// and the bar index from these — the true pivot bar, confirmBars back).
var int lastPHx = na
var int lastPLx = na
if not na(usePH)
    lastPHx := bar_index - confirmBars
if not na(usePL)
    lastPLx := bar_index - confirmBars

// ── ZigZag = market structure: each turning point is an HH/HL/LH/LL ──
// A same-direction more-extreme pivot EXTENDS the current point (its label
// re-evaluates); an opposite pivot FLIPS to a new leg — only if it clears the
// leg-size filter above.
if not na(usePH)
    if zzD1 == 1
        if usePH > zzP1                                       // higher high → extend up-leg
            zzP1 := usePH
            zzX1 := lastPHx
            zzHigh := usePH
            zzLegEvent := true
            if showZigZag and not na(zzSeg)
                line.set_xy2(zzSeg, zzX1, zzP1)
            if not na(zzLbl)
                label.set_xy(zzLbl, zzX1, zzP1)
                label.set_text(zzLbl, na(zzPrevHigh) or zzHigh > zzPrevHigh ? "HH" : "LH")
    else if na(zzP1)                                          // very first point
        zzP1 := usePH
        zzX1 := lastPHx
        zzD1 := 1
        zzHigh := usePH
    else if math.abs(usePH - zzP1) >= zzMinLeg                // flip low→high (big enough)
        zzP0 := zzP1
        zzX0 := zzX1
        zzP1 := usePH
        zzX1 := lastPHx
        zzD1 := 1
        zzPrevHigh := zzHigh
        zzHigh := usePH
        zzLegEvent := true
        if showZigZag
            zzSeg := line.new(zzX0, zzP0, zzX1, zzP1, color = zzCol, width = zzWidth)
        if showStruct
            zzLbl := label.new(zzX1, zzP1, na(zzPrevHigh) or zzHigh > zzPrevHigh ? "HH" : "LH", style = label.style_label_down, color = color.new(bearCol, 70), textcolor = dashTextCol, size = lblSize)

if not na(usePL)
    if zzD1 == -1
        if usePL < zzP1                                       // lower low → extend down-leg
            zzP1 := usePL
            zzX1 := lastPLx
            zzLow := usePL
            zzLegEvent := true
            if showZigZag and not na(zzSeg)
                line.set_xy2(zzSeg, zzX1, zzP1)
            if not na(zzLbl)
                label.set_xy(zzLbl, zzX1, zzP1)
                label.set_text(zzLbl, na(zzPrevLow) or zzLow < zzPrevLow ? "LL" : "HL")
    else if na(zzP1)
        zzP1 := usePL
        zzX1 := lastPLx
        zzD1 := -1
        zzLow := usePL
    else if math.abs(usePL - zzP1) >= zzMinLeg                // flip high→low (big enough)
        zzP0 := zzP1
        zzX0 := zzX1
        zzP1 := usePL
        zzX1 := lastPLx
        zzD1 := -1
        zzPrevLow := zzLow
        zzLow := usePL
        zzLegEvent := true
        if showZigZag
            zzSeg := line.new(zzX0, zzP0, zzX1, zzP1, color = zzCol, width = zzWidth)
        if showStruct
            zzLbl := label.new(zzX1, zzP1, na(zzPrevLow) or zzLow < zzPrevLow ? "LL" : "HL", style = label.style_label_up, color = color.new(bullCol, 70), textcolor = dashTextCol, size = lblSize)

// Latest ZigZag leg (zzP0 → zzP1) → the Fib impulse the Golden Zone retraces.
validLeg = zzD1 != 0 and not na(zzP0) and not na(zzP1) and zzP0 != zzP1
legBull  = zzD1 == 1                              // up-leg ⇒ bullish retracement zone
legHigh  = validLeg ? math.max(zzP0, zzP1) : na
legLow   = validLeg ? math.min(zzP0, zzP1) : na
legHx    = zzD1 == 1 ? zzX1 : zzX0               // bar index of the leg's high / low
legLx    = zzD1 == 1 ? zzX0 : zzX1

// Bias from the ZigZag structure: HH+HL ⇒ bullish, LH+LL ⇒ bearish, else neutral.
if not na(zzHigh) and not na(zzPrevHigh) and not na(zzLow) and not na(zzPrevLow)
    bias := (zzHigh > zzPrevHigh and zzLow > zzPrevLow) ? "Bullish" : (zzHigh < zzPrevHigh and zzLow < zzPrevLow) ? "Bearish" : "Neutral"

// ─────────────────────────────────────────────────────────────────────────────
// 4. CANDIDATE EVALUATION (direction, freshness, poison guard)
// ─────────────────────────────────────────────────────────────────────────────
// Direction: the latest ZigZag leg decides (up-leg ⇒ bullish retracement zone).
dirBull = legBull

// A candidate is structurally valid only with a real leg of positive range.
validSetup = validLeg and (legHigh > legLow)

// Active-setup state, held in series vars so signals can read them every bar.
var bool  aBull   = false   // active direction (meaningful only when aSet)
var bool  aSet    = false   // an active setup exists (cleared when superseded)
var bool  aAlive  = false   // false once invalidated or superseded → silent
var bool  aEntered  = false // once-per-setup latches (see alert engine)
var bool  aRejected = false
var bool  aBroke    = false
var bool  aHit1272  = false
var bool  aHit1618  = false
var float aHigh   = na
var float aLow    = na
var int   aHx     = na
var int   aLx     = na
var int   aBornBar = na     // signals become eligible AFTER this bar
var float aInvBuf  = na     // invalidation ATR buffer, FROZEN at creation (repair #5)
var string zoneStatus = "Waiting"

// Poison guard (repair #1): when a setup is invalidated, its leg identity
// (direction + start-pivot bar) is recorded. A later extension of that SAME
// leg cannot spawn a fresh setup — the structure is already broken. A new leg
// (different start pivot) evaluates normally.
var bool legPoisoned  = false
var bool poisonBull   = false
var int  poisonAnchor = na

// ── True late-zone detection (repair #2) ──
// Scan EVERY bar from the actual pivot through its confirmation bar
// (confirmBars + 1 bars). Computed unconditionally on every bar so history
// stays consistent. Wick mode uses highs/lows; Close mode uses closes.
winLen        = confirmBars + 1
lowLowWin     = ta.lowest(low,   winLen)
highHighWin   = ta.highest(high, winLen)
lowCloseWin   = ta.lowest(close,  winLen)
highCloseWin  = ta.highest(close, winLen)

// Near edge = the smaller golden ratio — the first level a retrace meets.
nearRatio = math.min(goldenLower, goldenUpper)
cRng  = nz(legHigh) - nz(legLow)
cNear = dirBull ? nz(legHigh) - nearRatio * cRng : nz(legLow) + nearRatio * cRng
// STRICT FRESHNESS (documented choice): equality counts as late. If price
// already TOUCHED the near edge anywhere in the pivot→confirmation window,
// the reaction has started and the candidate is late.
bullLate  = touchWick ? lowLowWin   <= cNear : lowCloseWin  <= cNear
bearLate  = touchWick ? highHighWin >= cNear : highCloseWin >= cNear
lateZone  = dirBull ? bullLate : bearLate
zoneFresh = not skipLateZones or not lateZone

// ── Candidate → outcome (repair #1) ──
candidateEvent = zzLegEvent and validSetup
poisonedCand   = legPoisoned and dirBull == poisonBull and (dirBull ? legLx : legHx) == poisonAnchor
newSetup       = candidateEvent and zoneFresh and not poisonedCand
skippedSetup   = candidateEvent and (not zoneFresh or poisonedCand)

// ─────────────────────────────────────────────────────────────────────────────
// 5. DRAWING OBJECTS (single active set; history is frozen/orphaned)
// ─────────────────────────────────────────────────────────────────────────────
var line[]  fLines  = array.new<line>()
var label[] fLabels = array.new<label>()
var box     gBox    = na

if candidateEvent
    // Any candidate — accepted OR skipped — supersedes the previous setup's
    // transient drawings. `for ... in` no-ops on empty arrays.
    for ln in fLines
        line.delete(ln)
    for lb in fLabels
        label.delete(lb)

    // Same leg, new extreme? Then the old box is a provisional draft of THIS
    // zone — replace it instead of archiving it (repair #9, default ON).
    // (Bull leg: the anchor is the leg's start LOW; bear leg: its start HIGH.)
    // NOTE: aBull/aLx/aHx still hold the PREVIOUS setup here (reassigned below).
    sameLeg = oneZonePerLeg and aSet and dirBull == aBull and (dirBull ? legLx == aLx : legHx == aHx)

    if not showHist or sameLeg
        if not na(gBox)
            box.delete(gBox)        // replace: remove the previous zone/draft
    else if not na(gBox)
        box.set_right(gBox, bar_index)                              // freeze …
        box.set_bgcolor(gBox, color.new(aBull ? bullZoneCol : bearZoneCol, 87))
        box.set_border_color(gBox, color.new(aBull ? color.green : color.red, 65))
        box.set_text_color(gBox, color.new(dashTextCol, 55))        // … and soften
    array.clear(fLines)
    array.clear(fLabels)
    gBox := na

    if newSetup
        aBull  := dirBull
        aSet   := true
        aAlive := true
        aEntered  := false           // all alert latches re-arm
        aRejected := false
        aBroke    := false
        aHit1272  := false
        aHit1618  := false
        aHigh  := legHigh
        aLow   := legLow
        aHx    := legHx
        aLx    := legLx
        aBornBar := bar_index        // signals eligible from the NEXT bar
        // Repair #5: freeze the invalidation buffer NOW. The protective price
        // (aLow − aInvBuf for bulls, aHigh + aInvBuf for bears) stays fixed
        // for this setup's whole lifetime — later ATR changes cannot move it.
        // nz(): during ATR warm-up the buffer degrades to 0 (strict), which is
        // the SAFE direction (never widens the stop zone unexpectedly).
        aInvBuf := invBufAtr * nz(atrZZ)
        zoneStatus := "Waiting"

        // Drawings may not sit more than 10,000 bars in the past — clamp the
        // left anchor so an extreme leg can never crash the script at runtime.
        leftX  = math.max(math.min(aHx, aLx), bar_index - 9500)
        rightX = bar_index + extendBars
        accent = aBull ? bullCol : bearCol

        // --- Fib retracement levels (0 … 1.0) ---
        if showFib
            for i = 0 to array.size(retraceRatios) - 1
                r  = array.get(retraceRatios, i)
                p  = f_retracement_price(aBull, aHigh, aLow, r)
                array.push(fLines, line.new(leftX, p, rightX, p, color = color.new(fibCol, 0), width = lineWidth, extend = extend.none))
                array.push(fLabels, label.new(rightX, p, str.format("{0,number,#.###}  ({1})", r, str.tostring(p, format.mintick)), style = label.style_label_left, color = color.new(accent, 80), textcolor = dashTextCol, size = lblSize))
            // --- Continuation projections (separately labelled, repair #8) ---
            for i = 0 to array.size(projMults) - 1
                m  = array.get(projMults, i)
                p  = f_projection_price(aBull, aHigh, aLow, m)
                array.push(fLines, line.new(leftX, p, rightX, p, color = color.new(fibCol, 0), width = lineWidth, extend = extend.none))
                array.push(fLabels, label.new(rightX, p, str.format("{0,number,#.###}× Projection  ({1})", m, str.tostring(p, format.mintick)), style = label.style_label_left, color = color.new(accent, 80), textcolor = dashTextCol, size = lblSize))

        // --- Golden Zone fill (title lives inside the box — no extra label) ---
        if showGolden
            gz1 = f_retracement_price(aBull, aHigh, aLow, goldenLower)
            gz2 = f_retracement_price(aBull, aHigh, aLow, goldenUpper)
            gTop = math.max(gz1, gz2)
            gBot = math.min(gz1, gz2)
            zoneFill   = aBull ? bullZoneCol : bearZoneCol
            zoneAccent = color.new(aBull ? color.green : color.red, 40)
            gBox := box.new(leftX, gTop, rightX, gBot, border_color = zoneAccent, bgcolor = zoneFill, text = aBull ? "GOLDEN ZONE ▲" : "GOLDEN ZONE ▼", text_color = color.new(dashTextCol, 25), text_size = lblSize, text_halign = text.align_left, text_valign = text.align_center)
    else
        // skippedSetup (repair #1): the candidate was late or its leg is
        // poisoned. The PREVIOUS setup is retired immediately — silently.
        // No "Invalidated" alert (price never broke it); no further touch /
        // rejection / breakout / target alerts can come from it.
        if aSet and aAlive
            zoneStatus := "Superseded"
        aAlive := false
        aSet   := false

// ─────────────────────────────────────────────────────────────────────────────
// 6. EXTEND ACTIVE SET TO THE RIGHT EDGE (only the active set is referenced)
// ─────────────────────────────────────────────────────────────────────────────
if barstate.islast and aSet
    rX = bar_index + extendBars
    for ln in fLines
        line.set_x2(ln, rX)
    for lb in fLabels
        label.set_x(lb, rX)
    if not na(gBox)
        box.set_right(gBox, rX)

// ─────────────────────────────────────────────────────────────────────────────
// 7. ZONE PRICES + RAW EVENT DETECTION
// ─────────────────────────────────────────────────────────────────────────────
rngA   = aHigh - aLow
gA     = f_retracement_price(aBull, aHigh, aLow, goldenLower)
gB     = f_retracement_price(aBull, aHigh, aLow, goldenUpper)
gLowP  = math.min(gA, gB)
gHighP = math.max(gA, gB)

// A setup only produces signals while ALIVE: not invalidated, not superseded.
activeValid = aSet and aAlive and not na(aHigh) and not na(aLow) and rngA > 0

// Raw conditions (drive the visual status; the alert engine below arbitrates).
touchedZone = activeValid and (touchWick ? (low <= gHighP and high >= gLowP) : (close <= gHighP and close >= gLowP))
prevInside  = close[1] <= gHighP and close[1] >= gLowP
bullRejectRaw = activeValid and aBull     and (touchWick ? low  <= gHighP : prevInside) and close > gHighP and close > open
bearRejectRaw = activeValid and not aBull and (touchWick ? high >= gLowP  : prevInside) and close < gLowP  and close < open
breakoutRaw   = activeValid and (aBull ? close > gHighP and close[1] <= gHighP : close < gLowP and close[1] >= gLowP)
// Invalidation boundary is FIXED: swing ± the buffer frozen at creation.
invalidatedRaw = activeValid and (aBull ? close < aLow - aInvBuf : close > aHigh + aInvBuf)
// Continuation projections.
extP1272 = f_projection_price(aBull, aHigh, aLow, 1.272)
extP1618 = f_projection_price(aBull, aHigh, aLow, 1.618)
reach1272Raw = activeValid and (aBull ? high >= extP1272 : low <= extP1272)
reach1618Raw = activeValid and (aBull ? high >= extP1618 : low <= extP1618)

// ─────────────────────────────────────────────────────────────────────────────
// 8. DETERMINISTIC ALERT ENGINE (repairs #3 + #4)
// ─────────────────────────────────────────────────────────────────────────────
// • At most ONE alert per candle, priority: Invalidation > 1.618 > 1.272 >
//   Rejection > Breakout > Entry.
// • Every event fires at most once per setup (latches consult RAW conditions,
//   so implied events complete too: a rejection on the first-touch candle
//   also completes the entry latch; reaching 1.618 directly completes 1.272).
// • No zone-interaction alert on the candle that CREATES a setup — signals
//   are eligible from the following confirmed candle (aBornBar).
// • barstate.isconfirmed is baked into every final alert boolean — alerts can
//   only become true on a confirmed candle, regardless of the user's alert
//   frequency setting.
sigEligible = activeValid and not na(aBornBar) and bar_index > aBornBar
sigOK       = barstate.isconfirmed and sigEligible

evInvalid = sigOK and invalidatedRaw
ev1618    = sigOK and not evInvalid and reach1618Raw and not aHit1618
ev1272    = sigOK and not evInvalid and not ev1618 and reach1272Raw and not aHit1272
evBullRej = sigOK and not evInvalid and not ev1618 and not ev1272 and bullRejectRaw and not aRejected
evBearRej = sigOK and not evInvalid and not ev1618 and not ev1272 and bearRejectRaw and not aRejected
evBreak   = sigOK and not evInvalid and not ev1618 and not ev1272 and not evBullRej and not evBearRej and breakoutRaw and not aBroke
evEntry   = sigOK and not evInvalid and not ev1618 and not ev1272 and not evBullRej and not evBearRej and not evBreak and touchedZone and not aEntered

// Commit latches AFTER the events were selected (this bar's alerts see the
// pre-commit latch state; from the next bar the latches suppress repeats).
if sigOK
    if touchedZone
        aEntered := true
    if bullRejectRaw or bearRejectRaw
        aRejected := true
    if breakoutRaw
        aBroke := true
    if reach1272Raw
        aHit1272 := true
    if reach1618Raw
        aHit1272 := true             // 1.618 direct → 1.272 completes too
        aHit1618 := true
    if invalidatedRaw
        aAlive := false              // permanent: the setup is dead
        legPoisoned  := true         // …and its leg cannot spawn a new setup
        poisonBull   := aBull
        poisonAnchor := aBull ? aLx : aHx
        if not na(gBox)              // gray the dead zone out immediately
            box.set_bgcolor(gBox, color.new(color.gray, 88))
            box.set_border_color(gBox, color.new(color.gray, 60))
            box.set_text(gBox, "GOLDEN ZONE ✕")
            box.set_text_color(gBox, color.new(dashTextCol, 60))

// Persistent zone status string (visual readout; may update intrabar, alerts
// never do).
if activeValid
    if invalidatedRaw
        zoneStatus := "Invalidated"
    else if bullRejectRaw or bearRejectRaw
        zoneStatus := "Rejected"
    else if breakoutRaw
        zoneStatus := "Broke Out"
    else if touchedZone
        zoneStatus := "Inside Zone"

// ─────────────────────────────────────────────────────────────────────────────

// ─────────────────────────────────────────────────────────────────────────────
alertcondition(evEntry,   "Price Enters Golden Zone",  "Price entered the Golden Zone")
alertcondition(evBullRej, "Bullish Rejection",         "Bullish rejection from the Golden Zone")
alertcondition(evBearRej, "Bearish Rejection",         "Bearish rejection from the Golden Zone")
alertcondition(evBreak,   "Zone Breakout",             "Price broke straight through the Golden Zone")
alertcondition(evInvalid, "Golden Zone Invalidated",   "Golden Zone setup invalidated")
alertcondition(ev1272,    "1.272× Projection Reached", "Price reached the 1.272× impulse projection")
alertcondition(ev1618,    "1.618× Projection Reached", "Price reached the 1.618× impulse projection")
