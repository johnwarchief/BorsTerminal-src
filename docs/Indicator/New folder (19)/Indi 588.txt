f_add_swing(array<float> prices, array<int> bars, array<int> sides, float price, int bi, int side) =>
    if prices.size() > 0 and sides.get(0) == side
        bool replace = side == 1 ? price >= prices.get(0) : price <= prices.get(0)
        if replace
            prices.set(0, price)
            bars.set(0, bi)
    else
        prices.unshift(price)
        bars.unshift(bi)
        sides.unshift(side)

    while prices.size() > swingMemory
        prices.pop()
        bars.pop()
        sides.pop()
    true

f_find_range(array<float> prices, array<int> bars, array<int> sides, float atrValue) =>
    bool found = false
    float bestTop = na
    float bestBottom = na
    int bestLeft = na
    int bestHighTouches = 0
    int bestLowTouches = 0
    int n = prices.size()
    int minEvents = math.max(minTouches * 2, minRotations + 1)
    float tolerance = atrValue * clusterATR

    if n >= minEvents and atrValue > 0
        for count = minEvents to n
            float highSum = 0.0
            float lowSum = 0.0
            float highMin = na
            float highMax = na
            float lowMin = na
            float lowMax = na
            int highCount = 0
            int lowCount = 0
            int oldest = bar_index
            int newest = 0

            for i = 0 to count - 1
                float p = prices.get(i)
                int bi = bars.get(i)
                int side = sides.get(i)

                oldest := math.min(oldest, bi)
                newest := math.max(newest, bi)

                if side == 1
                    highSum += p
                    highCount += 1
                    highMin := na(highMin) ? p : math.min(highMin, p)
                    highMax := na(highMax) ? p : math.max(highMax, p)
                else
                    lowSum += p
                    lowCount += 1
                    lowMin := na(lowMin) ? p : math.min(lowMin, p)
                    lowMax := na(lowMax) ? p : math.max(lowMax, p)

            if highCount >= minTouches and lowCount >= minTouches
                float top = highSum / highCount
                float bottom = lowSum / lowCount
                float width = top - bottom
                bool stable = highMax - highMin <= tolerance * 2.0 and lowMax - lowMin <= tolerance * 2.0
                bool widthOK = width >= atrValue * minWidthATR and width <= atrValue * math.max(maxWidthATR, minWidthATR)
                bool spanOK = newest - oldest >= minRangeBars
                bool rotationsOK = count - 1 >= minRotations

                if top > bottom and stable and widthOK and spanOK and rotationsOK
                    found := true
                    bestTop := top
                    bestBottom := bottom
                    bestLeft := oldest
                    bestHighTouches := highCount
                    bestLowTouches := lowCount

    [found, bestTop, bestBottom, bestLeft, bestHighTouches, bestLowTouches]

f_reset_profile(array<float> profile) =>
    profile.clear()
    for i = 0 to profileBins - 1
        profile.push(0.0)
    true

f_add_profile_bar(array<float> profile, float barLow, float barHigh, float barVolume, float bottom, float top) =>
    float height = top - bottom

    if height > 0 and barVolume > 0
        if barHigh == barLow
            if barHigh >= bottom and barHigh <= top
                int idx = int(math.floor((barHigh - bottom) / height * profileBins))
                idx := math.max(0, math.min(profileBins - 1, idx))
                profile.set(idx, profile.get(idx) + barVolume)
        else
            float candleSpan = math.max(barHigh - barLow, syminfo.mintick)

            for i = 0 to profileBins - 1
                float binLow = bottom + height * i / profileBins
                float binHigh = bottom + height * (i + 1) / profileBins
                float overlap = math.max(0.0, math.min(barHigh, binHigh) - math.max(barLow, binLow))

                if overlap > 0
                    profile.set(i, profile.get(i) + barVolume * overlap / candleSpan)
    true

f_rebuild_profile(array<float> profile, int left, float top, float bottom) =>
    f_reset_profile(profile)
    int span = math.min(bar_index - left, 1500)

    if span >= 0
        for i = 0 to span
            f_add_profile_bar(profile, low[i], high[i], nz(volume[i], 0.0), bottom, top)
    true

    array<box> gradient = array.new<box>()

    line upper = line.new(left, top, bar_index, top, xloc=xloc.bar_index, color=na, width=2)
    line lower = line.new(left, bottom, bar_index, bottom, xloc=xloc.bar_index, color=na, width=2)

    line poc = line.new(left, math.avg(top, bottom), bar_index, math.avg(top, bottom), xloc=xloc.bar_index, color=na, width=2)

    RangeDraw.new(shell, gradient, profile, upper, lower, middle, poc, na, na, na, na, na, na, na, na, na, na, na, na)

f_render_profile(RangeDraw d, array<float> profile, int left, int right, float top, float bottom, color css, bool completed) =>
    float maxVol = 0.0
    int pocIndex = 0

    if profile.size() > 0
        for i = 0 to profile.size() - 1
            float v = profile.get(i)
            if v > maxVol
                maxVol := v
                pocIndex := i

    int totalBars = math.max(right - left + 1, 1)
    int maxProfileBars = math.max(2, int(math.round(totalBars * profileWidth / 100.0)))
    maxProfileBars := math.min(maxProfileBars, totalBars)
    float height = top - bottom

    if d.profile.size() > 0
        for i = 0 to d.profile.size() - 1
            box b = d.profile.get(i)
            float v = profile.size() > i ? profile.get(i) : 0.0
            float ratio = maxVol > 0 ? v / maxVol : 0.0
            float density = math.pow(ratio, 0.72)
            float binBottom = bottom + height * i / profileBins
            float binTop = bottom + height * (i + 1) / profileBins

            if showProfile and ratio > 0
                int widthBars = math.max(1, int(math.round(maxProfileBars * math.pow(ratio, 0.85))))
                int alpha = int(math.max(42.0, (completed ? 94.0 : 91.0) - density * (completed ? 30.0 : 42.0)))
                b.set_lefttop(right - widthBars + 1, binTop)
                b.set_rightbottom(right, binBottom)
                b.set_bgcolor(color.new(css, alpha))
            else
                b.set_bgcolor(na)

    float pocPrice = bottom + height * (pocIndex + 0.5) / profileBins
    d.poc.set_xy1(left, pocPrice)
    d.poc.set_xy2(right, pocPrice)
    d.poc.set_color(showPOC and maxVol > 0 ? color.new(chart.fg_color, completed ? 52 : 28) : na)
    true

// ┌────────────────────────────── Range Rendering ────────────────────────┐
// └──────────────────────────────────────────────────────────────────────────────────┘


f_render_range(RangeDraw d, array<float> profile, int left, int right, float top, float bottom, int highTouches, int lowTouches, color resolvedCol, bool completed) =>
    float height = math.max(top - bottom, syminfo.mintick)

    d.shell.set_lefttop(left, top)
    d.shell.set_rightbottom(right, bottom)
    d.shell.set_border_color(na)

    int rows = d.gradient.size()

    if rows > 0
        float rowHeight = height / rows

        for i = 0 to rows - 1
            float rowTop = top - rowHeight * i
            float rowBottom = rowTop - rowHeight
            float position = (i + 0.5) / rows
            float distance = math.abs(position - 0.5) * 2.0
            float strength = math.pow(distance, 1.55)

            color rowCol = position < 0.5 ? bearCol : bullCol
            int alpha = int(99.0 - strength * (completed ? 17.0 : 38.0))

            box layer = d.gradient.get(i)
            layer.set_lefttop(left, rowTop)
            layer.set_rightbottom(right, rowBottom)
            layer.set_bgcolor(color.new(rowCol, alpha))

    int upperAlpha = int(math.max(4.0, 43.0 - highTouches * 8.0))
    int lowerAlpha = int(math.max(4.0, 43.0 - lowTouches * 8.0))

    d.upper.set_xy1(left, top)
    d.upper.set_xy2(right, top)
    d.upper.set_color(color.new(bearCol, completed ? 34 : upperAlpha))

    d.lower.set_xy1(left, bottom)
    d.lower.set_xy2(right, bottom)
    d.lower.set_color(color.new(bullCol, completed ? 34 : lowerAlpha))

    float mid = math.avg(top, bottom)
    d.middle.set_xy1(left, mid)
    d.middle.set_xy2(right, mid)
    d.middle.set_color(showMidline ? color.new(chart.fg_color, 80) : na)

    f_render_profile(d, profile, left, right, top, bottom, completed ? resolvedCol : rangeCol, completed)
    true

    float tp2 = direction == 1 ? entry + height * tp2Mult : entry - height * tp2Mult
    float tp3 = direction == 1 ? entry + height * tp3Mult : entry - height * tp3Mult
    int right = breakBar + positionBars
    color css = direction == 1 ? bullCol : bearCol

    if showPosition
        if direction == 1
            d.rewardBox := box.new(breakBar, tp3, right, entry, xloc=xloc.bar_index, border_color=na, bgcolor=color.new(bullCol, 88))
            d.riskBox := box.new(breakBar, entry, right, stop, xloc=xloc.bar_index, border_color=na, bgcolor=color.new(bearCol, 93))

            d.rewardBox := box.new(breakBar, entry, right, tp3, xloc=xloc.bar_index, border_color=na, bgcolor=color.new(bearCol, 88))
            d.riskBox := box.new(breakBar, stop, right, entry, xloc=xloc.bar_index, border_color=na, bgcolor=color.new(bullCol, 93))

        highTouches := candidateHighs
        lowTouches := candidateLows
        activeDraw := f_new_range(rangeLeft, rangeTop, rangeBottom)

        f_rebuild_profile(rangeProfile, rangeLeft, rangeTop, rangeBottom)

        profileLastBar := bar_index
        breakUpCount := 0
        breakDownCount := 0
        deviationDir := 0
        deviationExtreme := na
        deviationExtremeBar := na

// ┌────────────────────────────── Refine Range ───────────────────────────┐
// └──────────────────────────────────────────────────────────────────────────────────┘

if rangeActive
    float tolerance = atr * clusterATR
    bool edgesChanged = false

    if not na(ph) and bar_index - swingLen >= rangeLeft and math.abs(ph - rangeTop) <= tolerance
        rangeTop += (ph - rangeTop) * edgeAdapt
        highTouches += 1
        edgesChanged := true

    if not na(pl) and bar_index - swingLen >= rangeLeft and math.abs(pl - rangeBottom) <= tolerance
        rangeBottom += (pl - rangeBottom) * edgeAdapt
        lowTouches += 1
        edgesChanged := true

    if edgesChanged
        f_rebuild_profile(rangeProfile, rangeLeft, rangeTop, rangeBottom)
        profileLastBar := bar_index
    else if bar_index > nz(profileLastBar, bar_index - 1)
        f_add_profile_bar(rangeProfile, low, high, nz(volume, 0.0), rangeBottom, rangeTop)
        profileLastBar := bar_index

// ┌────────────────────────────── Breaks & Deviations ────────────────────┐
// └──────────────────────────────────────────────────────────────────────────────────┘

int breakDirection = 0

if rangeActive and barstate.isconfirmed
    float buffer = atr * breakBuffer
    bool outsideUp = close > rangeTop + buffer
    bool outsideDown = close < rangeBottom - buffer

    breakUpCount := outsideUp ? breakUpCount + 1 : 0
    breakDownCount := outsideDown ? breakDownCount + 1 : 0

    if deviationDir == 0
        if outsideUp
            deviationDir := 1
            deviationExtreme := high
            deviationExtremeBar := bar_index
        else if outsideDown
            deviationDir := -1
            deviationExtreme := low
            deviationExtremeBar := bar_index
    else
        if deviationDir == 1 and high > deviationExtreme
            deviationExtreme := high
            deviationExtremeBar := bar_index

        if deviationDir == -1 and low < deviationExtreme
            deviationExtreme := low
            deviationExtremeBar := bar_index

        bool backInside = close <= rangeTop and close >= rangeBottom

        if backInside
            if showDeviations
                f_add_deviation(deviationExtremeBar, deviationExtreme, deviationDir)

            signalDeviationHigh := deviationDir == 1
            signalDeviationLow := deviationDir == -1
            deviationDir := 0
            deviationExtreme := na
            deviationExtremeBar := na

    if breakUpCount >= breakCloses
        breakDirection := 1

    if breakDownCount >= breakCloses
        breakDirection := -1

    if maxRangeAge > 0 and bar_index - rangeOpened >= maxRangeAge
        breakDirection := 2

// ┌────────────────────────────── Resolve Range ──────────────────────────┐
// └──────────────────────────────────────────────────────────────────────────────────┘

if rangeActive and breakDirection != 0
    color resolvedCol = breakDirection == 1 ? bullCol : breakDirection == -1 ? bearCol : rangeCol

    f_render_range(activeDraw, rangeProfile, rangeLeft, bar_index, rangeTop, rangeBottom, highTouches, lowTouches, resolvedCol, true)

    if breakDirection == 1 or breakDirection == -1
        bool bullishBreak = breakDirection == 1
        float entry = bullishBreak ? rangeTop : rangeBottom

        if showBreakLabel
            activeDraw.breakTag := label.new(bar_index, entry, bullishBreak ? "BREAKOUT  ↑" : "BREAKDOWN  ↓", xloc=xloc.bar_index, yloc=yloc.price, style=label.style_label_left, color=color.new(resolvedCol, 88), textcolor=chart.fg_color, size=size.small)

        f_draw_position(activeDraw, breakDirection, bar_index, rangeTop, rangeBottom)

        signalBreakUp := bullishBreak
        signalBreakDown := not bullishBreak

    history.unshift(activeDraw)

    while history.size() > historyCount
        RangeDraw old = history.pop()
        f_delete_range(old)

    rangeActive := false
    activeDraw := na

    swingPrice.clear()
    swingBar.clear()
    swingSide.clear()

    breakUpCount := 0
    breakDownCount := 0
    deviationDir := 0
    deviationExtreme := na
    deviationExtremeBar := na
    cooldown := cooldownBars

// ┌────────────────────────────── Active Rendering ───────────────────────┐
// └──────────────────────────────────────────────────────────────────────────────────┘

if rangeActive
    f_render_range(activeDraw, rangeProfile, rangeLeft, bar_index, rangeTop, rangeBottom, highTouches, lowTouches, rangeCol, false)

// ┌────────────────────────────── Candle Coloring ────────────────────────┐
// └──────────────────────────────────────────────────────────────────────────────────┘

color candleCol = na

if paintCandles and rangeActive and rangeTop > rangeBottom
    float pos = math.max(0.0, math.min(1.0, (close - rangeBottom) / (rangeTop - rangeBottom)))
    candleCol := pos <= 0.5 ? color.from_gradient(pos, 0.0, 0.5, bullCol, rangeCol) : color.from_gradient(pos, 0.5, 1.0, rangeCol, bearCol)
    candleCol := color.new(candleCol, candleTransparency)

    if close > rangeTop
        candleCol := bullCol
    else if close < rangeBottom
        candleCol := bearCol

barcolor(candleCol)

// ┌────────────────────────────── Data Window ────────────────────────────┐
// └──────────────────────────────────────────────────────────────────────────────────┘

plot(rangeActive ? rangeTop : na, "Active Range High", display=display.data_window)
plot(rangeActive ? rangeBottom : na, "Active Range Low", display=display.data_window)

alertcondition(signalBreakDown, "Range Breakdown", "Range Profile Structure broke downward | {{exchange}}:{{ticker}}")
