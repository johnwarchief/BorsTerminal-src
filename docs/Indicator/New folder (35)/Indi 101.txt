    float w1R = sh2 - sl2

    float w4Ret = w3R > 0 ? (sh1 - sl0) / w3R : 0

    bool bR1 = sl1 > sl2
    bool bR2 = w3R >= w1R * 0.7
    bool bR3 = sl0 > sl1 * 0.995

    bool bullImpulse = bR1 and bR2 and bR3 and bR4 and bR5 and bR6 and bTO and sh0 > sh2

    float bw1R = sh2 - sl2
    float bw3R = sh1 - sl1
    float bw2Ret = bw1R > 0 ? (sh1 - sl2) / bw1R : 0
    float bw4Ret = bw3R > 0 ? (sh0 - sl1) / bw3R : 0

    bool beR1 = sh1 < sh2
    bool beR2 = bw3R >= bw1R * 0.7
    bool beR3 = sh0 < sh1 * 1.005

    float rHigh    = array.get(shPrices, shSz - 1)
    int   rHighBar = array.get(shBars, shSz - 1)
    if rHighBar > wave5EndBar
        float bounce = rHigh - wave5End
        float iRng   = wave1Start - wave5End
        float aRet   = iRng > 0 ? bounce / iRng : 0
        if aRet > 0.20 and aRet < 0.72
            bearCorrNow := true
            abcActive   := true

// ═══════════════════════════════════════════════════════════════════════════════

var float fib382 = na
var float fib500 = na
var float fib618 = na
var float fibTop = na
var float fibBot = na

bool priceAtFib382 = false
bool priceAtFib500 = false
bool priceAtFib618 = false
float fibTol = atrVal * 0.6

if impulseActive and not na(wave5End) and not na(wave1Start)
    if waveTrend == 1
        fibTop := wave5End
        fibBot := wave1Start
    else
        fibTop := wave1Start
        fibBot := wave5End
    float fibRange = fibTop - fibBot
    fib382 := fibTop - fibRange * 0.382
    fib500 := fibTop - fibRange * 0.500
    fib618 := fibTop - fibRange * 0.618
    priceAtFib382 := math.abs(close - fib382) < fibTol
    priceAtFib500 := math.abs(close - fib500) < fibTol
    priceAtFib618 := math.abs(close - fib618) < fibTol

bool priceAtAnyFib = priceAtFib382 or priceAtFib500 or priceAtFib618

type FibZone
    box     bx
    float   top
    float   bottom
    int     startBar
    bool    isBull
    bool    active

var array<FibZone> fibZones = array.new<FibZone>()

if showFibBoxes and (bullImpulseNow or bearImpulseNow) and not na(fib382) and not na(fib618)
    if array.size(fibZones) >= fibMaxBoxes
        old = array.shift(fibZones)
        box.delete(old.bx)
    float zTop = math.max(fib382, fib618)
    float zBot = math.min(fib382, fib618)
    newZone = FibZone.new(
      bx=box.new(bar_index, zTop, bar_index + 5, zBot,
         border_color=color.new(FIB_COLOR, 40), bgcolor=color.new(FIB_DIM, 92),
         border_width=1, border_style=line.style_dotted, extend=extend.right),
      top=zTop, bottom=zBot, startBar=bar_index, isBull=waveTrend==1, active=true)
    array.push(fibZones, newZone)

var bool priceInBullFib = false
var bool priceInBearFib = false
priceInBullFib := false
priceInBearFib := false

if array.size(fibZones) > 0
    for i = array.size(fibZones) - 1 to 0
        fz = array.get(fibZones, i)
        if fz.active
            if bar_index - fz.startBar > fibMaxAge
                fz.active := false
                box.delete(fz.bx)
                array.remove(fibZones, i)
                continue
            if low <= fz.top and high >= fz.bottom
                if fz.isBull
                    priceInBullFib := true
                else
                    priceInBearFib := true
            filled = false
            if fz.isBull and close < fz.bottom - atrVal * 1.5

calcSLTP(bool isBull, float conf) =>
    float atrFloor = atrVal * slAtrMulti * 0.7
    float atrCap   = atrVal * slAtrMulti * 1.8
    float minRR1   = tp1AtrMulti / slAtrMulti
    float minRR2   = tp2AtrMulti / slAtrMulti
    float tp1Floor = atrVal * tp1AtrMulti * 0.8
    float tp2Floor = atrVal * tp2AtrMulti * 0.8

    // ── STOP LOSS: Structure-based ─────────────────────────────────────────────
    float sl = atrVal * slAtrMulti

    // Find nearest swing for structure-based SL
    if isBull and slSz >= 2
        float nearestSwingLow = na
        for j = slSz - 1 to math.max(0, slSz - 5)
            float sPrice = array.get(slPrices, j)
            int   sBar   = array.get(slBars, j)
            if sPrice < close and bar_index - sBar <= 50
                if na(nearestSwingLow) or sPrice > nearestSwingLow
                    nearestSwingLow := sPrice
        if not na(nearestSwingLow)
            float structSL = close - nearestSwingLow + atrVal * 0.3
            if structSL >= atrFloor and structSL <= atrCap
                sl := structSL

    if not isBull and shSz >= 2
        float nearestSwingHigh = na
        for j = shSz - 1 to math.max(0, shSz - 5)
            float sPrice = array.get(shPrices, j)
            int   sBar   = array.get(shBars, j)
            if sPrice > close and bar_index - sBar <= 50
                if na(nearestSwingHigh) or sPrice < nearestSwingHigh
                    nearestSwingHigh := sPrice
        if not na(nearestSwingHigh)
            float structSL = nearestSwingHigh - close + atrVal * 0.3
            if structSL >= atrFloor and structSL <= atrCap
                sl := structSL

    // Apply confidence modifier to SL
    float confMod = conf >= 70 ? 0.85 : conf >= 55 ? 0.92 : conf >= 40 ? 1.0 : 1.15
    sl := math.max(atrFloor, math.min(atrCap, sl * confMod))

    // ── TP1: Nearest structure target or Fibonacci ─────────────────────────────
    float t1 = sl * minRR1

    // Check nearest swing target
    if isBull and shSz >= 2
        float nearestTarget = na
        for j = shSz - 1 to math.max(0, shSz - 6)
            float sPrice = array.get(shPrices, j)
            int   sBar   = array.get(shBars, j)
            if sPrice > close and bar_index - sBar <= 80
                float dist = sPrice - close
                if dist > sl * 1.2
                    if na(nearestTarget) or dist < nearestTarget
                        nearestTarget := dist
        if not na(nearestTarget)
            t1 := nearestTarget

    if not isBull and slSz >= 2
        float nearestTarget = na
        for j = slSz - 1 to math.max(0, slSz - 6)
            float sPrice = array.get(slPrices, j)
            int   sBar   = array.get(slBars, j)
            if sPrice < close and bar_index - sBar <= 80
                float dist = close - sPrice
                if dist > sl * 1.2
                    if na(nearestTarget) or dist < nearestTarget
                        nearestTarget := dist
        if not na(nearestTarget)
            t1 := nearestTarget

    // Check Fibonacci levels as TP1 candidates
    if impulseActive and not na(fib382) and not na(fib500)
        float fibT1 = na
        if isBull
            if fib382 > close and fib382 - close > sl * 1.2
                fibT1 := fib382 - close
            else if fib500 > close and fib500 - close > sl * 1.2
                fibT1 := fib500 - close
        else
            if fib382 < close and close - fib382 > sl * 1.2
                fibT1 := close - fib382
            else if fib500 < close and close - fib500 > sl * 1.2
                fibT1 := close - fib500
        if not na(fibT1) and (fibT1 < t1 or t1 <= sl * minRR1)
            t1 := fibT1

    // Enforce minimum R:R and ATR floor
    t1 := math.max(t1, sl * minRR1)
    t1 := math.max(t1, tp1Floor)

    // ── TP2: Extended structure target, Fibonacci extension, or wave target ────
    float t2 = sl * minRR2

    // Wave-based target (Fibonacci extension of impulse)
    if impulseActive and not na(impulseRange) and impulseRange > 0
        float waveT2 = impulseRange * 0.618
        if waveT2 > t2
            t2 := waveT2

    // Check further swing targets
    if isBull and shSz >= 3
        float farTarget = na
        for j = shSz - 1 to math.max(0, shSz - 8)
            float sPrice = array.get(shPrices, j)
            int   sBar   = array.get(shBars, j)
            if sPrice > close and bar_index - sBar <= 120
                float dist = sPrice - close
                if dist > t1 * 1.3
                    if na(farTarget) or dist < farTarget
                        farTarget := dist
        if not na(farTarget)
            t2 := farTarget

    if not isBull and slSz >= 3
        float farTarget = na
        for j = slSz - 1 to math.max(0, slSz - 8)
            float sPrice = array.get(slPrices, j)
            int   sBar   = array.get(slBars, j)
            if sPrice < close and bar_index - sBar <= 120
                float dist = close - sPrice
                if dist > t1 * 1.3
                    if na(farTarget) or dist < farTarget
                        farTarget := dist
        if not na(farTarget)
            t2 := farTarget

    // Check Fibonacci extension for TP2
    if impulseActive and not na(fib618)
        float fibT2 = na
        if isBull and fib618 > close
            fibT2 := fib618 - close
        if not isBull and fib618 < close
            fibT2 := close - fib618
        if not na(fibT2) and fibT2 > t1 * 1.2 and fibT2 > t2
            t2 := fibT2

    // Enforce minimum R:R, ATR floor, and TP2 > TP1
    t2 := math.max(t2, sl * minRR2)
    t2 := math.max(t2, tp2Floor)
    t2 := math.max(t2, t1 * 1.5)
    
    [sl, t1, t2]

// Count active SLTP groups (each signal = 3 levels: SL + TP1 + TP2)
countActiveSLTP() =>
    int count = 0
    int lastBar = -1
    int sz = array.size(tradeLevels)
    if sz > 0
        for i = 0 to sz - 1
            tl = array.get(tradeLevels, i)
            if tl.active and tl.sigBar != lastBar
                count += 1
                lastBar := tl.sigBar
    count

// Helper: is SL already hit for this trade?
f_isSLHit(int sigBar) =>
    bool result = false
    int sz = array.size(tradeLevels)
    if sz > 0
        for k = 0 to sz - 1
            kl = array.get(tradeLevels, k)
            if kl.sigBar == sigBar and kl.levelType == 0 and kl.hit
                result := true
                break
    result

// Helper: is TP1 already hit for this trade?
f_isTP1Hit(int sigBar) =>
    bool result = false
    int sz = array.size(tradeLevels)
    if sz > 0
        for k = 0 to sz - 1
            kl = array.get(tradeLevels, k)
            if kl.sigBar == sigBar and kl.levelType == 1 and kl.hit
                result := true
                break
    result

if array.size(tradeLevels) > 0
    for i = 0 to array.size(tradeLevels) - 1
        tl = array.get(tradeLevels, i)
        if tl.active and not tl.hit

            bool hitNow = false
            if tl.isBull
                hitNow := tl.levelType == 0 ? low <= tl.price : high >= tl.price
            else
                hitNow := tl.levelType == 0 ? high >= tl.price : low <= tl.price

            if hitNow
                // ── SL hit — always processes, closes the trade ────────────
                if tl.levelType == 0
                    tl.hit := true
                    bool isBE = str.startswith(label.get_text(tl.lbl), "BE")
                    if isBE
                        label.set_color(tl.lbl,     color.new(NEUTRAL_COLOR, 15))
                        label.set_textcolor(tl.lbl, #ffffff)
                        label.set_text(tl.lbl,      label.get_text(tl.lbl) + " ✓")
                    else
                        label.set_color(tl.lbl,     color.new(BEAR_PRIMARY, 15))
                        label.set_textcolor(tl.lbl, #ffffff)

                // ── TP1 hit — only if SL not already hit ──────────────────
                else if tl.levelType == 1 and not f_isSLHit(tl.sigBar)
                    tl.hit := true
                    label.set_color(tl.lbl,     color.new(BULL_DIM, 15))
                    label.set_textcolor(tl.lbl, #1a1a1a)
                    // Move SL to break even if enabled
                    if moveToBreakEven
                        for j = 0 to array.size(tradeLevels) - 1
                            sl_tl = array.get(tradeLevels, j)
                            if sl_tl.sigBar == tl.sigBar and sl_tl.levelType == 0 and not sl_tl.hit
                                float bePrice = line.get_y2(sl_tl.vertLine)
                                sl_tl.price   := bePrice
                                line.set_y1(sl_tl.horizLine, bePrice)
                                line.set_y2(sl_tl.horizLine, bePrice)
                                line.set_y2(sl_tl.vertLine,  bePrice)
                                label.set_y(sl_tl.lbl,       bePrice)
                                label.set_text(sl_tl.lbl,    "BE " + str.tostring(bePrice, format.mintick))
                                label.set_color(sl_tl.lbl,     color.new(NEUTRAL_COLOR, 60))
                                label.set_textcolor(sl_tl.lbl, color.new(#ffffff, 40))
                                break

                // ── TP2 hit — only if SL not already hit ──────────────────
                else if tl.levelType == 2 and not f_isSLHit(tl.sigBar)
                    tl.hit := true
                    label.set_color(tl.lbl,     color.new(BULL_BRIGHT, 15))
                    label.set_textcolor(tl.lbl, #1a1a1a)
