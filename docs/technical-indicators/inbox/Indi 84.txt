safeDiv(float num, float den, float fallback = 0.0) =>
    den != 0 and not na(num) and not na(den) ? num / den : fallback

adaptiveDonchian(int basePeriod, float volRatio, float strength) =>
    int minP = math.max(math.round(basePeriod * (1.0 - strength)), 3)
    int maxP = math.round(basePeriod * (1.0 + strength))
    int adaptedLen = math.round(basePeriod * (1.0 + strength * (1.0 - 2.0 * volRatio)))
    int safePeriod = math.max(math.min(adaptedLen, maxP), minP)
    float hi = ta.highest(high, safePeriod)
    float lo = ta.lowest(low, safePeriod)
    math.avg(nz(hi, high), nz(lo, low))

classicDonchian(int len) =>
    float hi = ta.highest(high, len)
    float lo = ta.lowest(low, len)
    math.avg(nz(hi, high), nz(lo, low))

float momentumSmooth = nz(ta.ema(momentumPulse, 5), 0.0)

// Momentum state

bool momentumFading = momentumSmooth < nz(momentumSmooth[1], 0.0) and momentumSmooth > 0
bool momentumBearAccel = momentumSmooth < nz(momentumSmooth[1], 0.0) and momentumSmooth < 0

// ══════════════════════════════════════
// CONFLUENCE SCORING ENGINE (0–100)
// ══════════════════════════════════════
// 8 factors for v2.0

confluenceScore(bool isBull) =>
    float score = 0.0
    // Factor 1: Price vs Cloud (20 pts)
    score += isBull ? (priceAboveCloud ? 20.0 : priceInCloud ? 8.0 : 0.0) : (priceBelowCloud ? 20.0 : priceInCloud ? 8.0 : 0.0)
    // Factor 2: TK alignment (15 pts)
    score += isBull ? (tkBullish ? 15.0 : 0.0) : (tkBearish ? 15.0 : 0.0)
    // Factor 3: Chikou (15 pts)
    score += isBull ? (chikouBull ? 15.0 : 0.0) : (chikouBear ? 15.0 : 0.0)
    // Factor 4: Cloud direction (10 pts)
    float spanAslope = senkouA - nz(senkouA[3], senkouA)
    float spanBslope = senkouB - nz(senkouB[3], senkouB)
    bool futureCloudBull = spanAslope > spanBslope
    score += isBull ? (futureCloudBull ? 10.0 : 0.0) : (not futureCloudBull ? 10.0 : 0.0)
    // Factor 5: Kumo thickness (8 pts)
    score += not isThinKumo ? 8.0 : 2.0
    // Factor 6: Volume (10 pts)
    score += volConfirm ? 10.0 : 0.0
    // Factor 7: Momentum alignment (12 pts) — NEW
    score += isBull ? (momentumSmooth > 20 ? 12.0 : momentumSmooth > 0 ? 6.0 : 0.0) : (momentumSmooth < -20 ? 12.0 : momentumSmooth < 0 ? 6.0 : 0.0)
    // Factor 8: Momentum acceleration (10 pts) — NEW
    score += isBull ? (momentumRising ? 10.0 : 0.0) : (momentumBearAccel ? 10.0 : 0.0)
    math.min(score, 100.0)

// Calculate SL level based on mode
calcSL(bool isLong, float entry) =>
    float slLevel = 0.0
    float atrBuf = atrBase * slAtrMultInput
    if slModeInput == "Kumo Edge"
        slLevel := isLong ? kumoBot - atrBuf : kumoTop + atrBuf
    else if slModeInput == "Kijun"
        slLevel := isLong ? kijunSen - atrBuf : kijunSen + atrBuf
    else
        slLevel := isLong ? entry - atrBase * 2.0 : entry + atrBase * 2.0
    slLevel

// Check for new trade entry on any confirmed signal
bool anyBuy = confirmedBuy or (showBreakoutInput and kumoBreakUp) or (showBreakoutInput and retestBull)
bool anySell = confirmedSell or (showBreakoutInput and kumoBreakDown) or (showBreakoutInput and retestBear)

if anyBuy and showTPSLInput
    activeEntryPrice := close
    activeSL := calcSL(true, close)
    float riskDist = math.max(math.abs(close - activeSL), atrBase * 0.5)
    activeTP1 := close + riskDist * tp1MultInput
    activeTP2 := close + riskDist * tp2MultInput
    activeTP3 := close + riskDist * tp3MultInput
    activeDir := 1
    tp1Hit := false
    tp2Hit := false
    tp3Hit := false
    slHit := false

if anySell and showTPSLInput
    activeEntryPrice := close
    activeSL := calcSL(false, close)
    float riskDist = math.max(math.abs(activeSL - close), atrBase * 0.5)
    activeTP1 := close - riskDist * tp1MultInput
    activeTP2 := close - riskDist * tp2MultInput
    activeTP3 := close - riskDist * tp3MultInput
    activeDir := -1
    tp1Hit := false
    tp2Hit := false
    tp3Hit := false
    slHit := false

// Track TP/SL hits (with na guards for safety)
// Trade stays open until TP3 or SL is reached
if activeDir == 1 and not slHit and not na(activeSL)
    if not na(activeTP1) and high >= activeTP1 and not tp1Hit
        tp1Hit := true
    if not na(activeTP2) and high >= activeTP2 and not tp2Hit
        tp2Hit := true
    if not na(activeTP3) and high >= activeTP3 and not tp3Hit
        tp3Hit := true
        activeDir := 0
    if low <= activeSL
        slHit := true
        activeDir := 0

if activeDir == -1 and not slHit and not na(activeSL)
    if not na(activeTP1) and low <= activeTP1 and not tp1Hit
        tp1Hit := true
    if not na(activeTP2) and low <= activeTP2 and not tp2Hit
        tp2Hit := true
    if not na(activeTP3) and low <= activeTP3 and not tp3Hit
        tp3Hit := true
        activeDir := 0
    if high >= activeSL
        slHit := true
        activeDir := 0

// ══════════════════════════════════════
// MTF CONFLUENCE
// ══════════════════════════════════════
// Each TF: single request.security returning pre-computed state int
// Formula: tenkan > kijun AND close > max(spanA, spanB) → +1 (bull)
//          tenkan < kijun AND close < min(spanA, spanB) → -1 (bear)
//          else → 0 (neutral)

mtfCalcState() =>
    float t = classicDonchian(conversionBaseInput)
    float k = classicDonchian(baseBaseInput)
    float sa = math.avg(t, k)
    float sb = classicDonchian(spanBBaseInput)
    float c = close
    float hi = math.max(sa, sb)
    float lo = math.min(sa, sb)
    int st = c > hi and t > k ? 1 : c < lo and t < k ? -1 : 0
    st
