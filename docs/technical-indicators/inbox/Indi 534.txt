get_name(string full, string short) =>
    show_indis == "Full" ? full : (show_indis == "First Letter" ? short : "")

method validate_slope(series float src, series float ind, int c_prd, int offset, int len, float p_price, float p_ind, bool is_low) =>
    bool is_clean = true
    float price_slope = (src[c_prd] - p_price) / len
    float ind_slope = (ind[c_prd] - p_ind) / len
    for j = 1 to len - 1
        int check_idx = c_prd + j
        float expected_price = src[c_prd] - (price_slope * j)
        float expected_ind = ind[c_prd] - (ind_slope * j)
        if is_low
            if src[check_idx] < expected_price or ind[check_idx] < expected_ind
                is_clean := false
                break
        else
            if src[check_idx] > expected_price or ind[check_idx] > expected_ind
                is_clean := false
                break
    is_clean

method check_div(series float src, series float ind, bool is_ph, bool is_pl, array<Pivot> p_highs, array<Pivot> p_lows, int c_prd, int max_b, bool check_reg, bool check_hid) =>
    bool pos_reg = false, bool neg_reg = false, bool pos_hid = false, bool neg_hid = false
    int opr = 0, int onr = 0, int oph = 0, int onh = 0
    if is_pl
        for p in p_lows
            int curr_bar = bar_index - c_prd
            int len = curr_bar - p.bar
            if len > max_b
                break
            if len > 5
                int offset = bar_index - p.bar
                float p_ind = ind[offset]
                if check_reg and src[c_prd] < p.price and ind[c_prd] > p_ind
                    if validate_slope(src, ind, c_prd, offset, len, p.price, p_ind, true)
                        pos_reg := true, opr := offset
                        break
                if check_hid and src[c_prd] > p.price and ind[c_prd] < p_ind
                    if validate_slope(src, ind, c_prd, offset, len, p.price, p_ind, true)
                        pos_hid := true, oph := offset
                        break
    if is_ph
        for p in p_highs
            int curr_bar = bar_index - c_prd
            int len = curr_bar - p.bar
            if len > max_b
                break
            if len > 5
                int offset = bar_index - p.bar
                float p_ind = ind[offset]
                if check_reg and src[c_prd] > p.price and ind[c_prd] < p_ind
                    if validate_slope(src, ind, c_prd, offset, len, p.price, p_ind, false)
                        neg_reg := true, onr := offset
                        break
                if check_hid and src[c_prd] < p.price and ind[c_prd] > p_ind
                    if validate_slope(src, ind, c_prd, offset, len, p.price, p_ind, false)
                        neg_hid := true, onh := offset
                        break
    [pos_reg, neg_reg, pos_hid, neg_hid, opr, onr, oph, onh]

method add_signal(array<DivSignal> arr, int type, int offset, string name, color col) =>
    bool found = false
    for s in arr
        if s.type == type and s.offset == offset
            if name != ""
                s.txt += (s.txt == "" ? "" : "\n") + name
            s.count += 1
            found := true
            break
    if not found
        arr.push(DivSignal.new(type, offset, name, col, 1))

method process_indicator(array<DivSignal> sigs, bool enabled, string name, series float src, series float ind, bool is_ph, bool is_pl, array<Pivot> phs, array<Pivot> pls, int c_prd, int mx, bool c_reg, bool c_hid) =>
    if enabled
        [pr, nr, ph, nh, opr, onr, oph, onh] = check_div(src, ind, is_ph, is_pl, phs, pls, c_prd, mx, c_reg, c_hid)
        if pr
            sigs.add_signal(0, opr, name, pos_reg_col)
        if nr
            sigs.add_signal(1, onr, name, neg_reg_col)
        if ph
            sigs.add_signal(2, oph, name, pos_hid_col)
        if nh
            sigs.add_signal(3, onh, name, neg_hid_col)

// --- SMC METHODS ---
getAutoHTF() =>
    bool isMins = timeframe.isminutes, bool isDaily = timeframe.isdaily, bool isWeekly = timeframe.isweekly
    int mult = timeframe.multiplier
    string htf = "1D" 
    if isMins
        if mult < 60
            htf := "60"
        else if mult >= 60 and mult <= 180
            htf := "240"
        else
            htf := "1D"
    else if isDaily
        htf := "1W"
    else if isWeekly
        htf := "1M"
    else
        htf := "1M"
    htf

calc_smc(int pLeft, int pRight, int lookback) =>
    float smc_ph = ta.pivothigh(high, pLeft, pRight)
    float smc_pl = ta.pivotlow(low, pLeft, pRight)
    var float lastSwingHighPrice = na, var int lastSwingHighTime = na
    var float lastSwingLowPrice = na, var int lastSwingLowTime = na

    if not na(smc_ph)
        lastSwingHighPrice := high[pRight]
        lastSwingHighTime  := time[pRight]
    if not na(smc_pl)
        lastSwingLowPrice := low[pRight]
        lastSwingLowTime  := time[pRight]

    var int trendDir = 1
    int activeDir = 0, float chochLvl = na, int chochT = na
    bool bearCHoCH = false, bool bullCHoCH = false

    if trendDir == 1 and not na(lastSwingLowPrice)
        if close < lastSwingLowPrice 
            trendDir := -1, chochLvl := lastSwingLowPrice, chochT := lastSwingLowTime, bearCHoCH := true
    else if trendDir == -1 and not na(lastSwingHighPrice)
        if close > lastSwingHighPrice 
            trendDir := 1, chochLvl := lastSwingHighPrice, chochT := lastSwingHighTime, bullCHoCH := true

    float oTop = na, float oBot = na, int oTime = na
    float fTop = na, float fBot = na, int fTime = na

    if bearCHoCH
        bool fFVG = false, bool fOB = false
        for i = 1 to lookback
            if high[i] < low[i+2] and close[i+1] < open[i+1]
                fTop := low[i+2], fBot := high[i], fTime := time[i+2]
                fFVG := true
                break
        for i = 1 to lookback
            if close[i] > open[i] and time[i] <= lastSwingHighTime
                oTop := high[i], oBot := low[i], oTime := time[i]
                fOB := true
                break
        if fFVG and fOB
            activeDir := -1

    if bullCHoCH
        bool fFVG = false, bool fOB = false
        for i = 1 to lookback
            if low[i] > high[i+2] and close[i+1] > open[i+1]
                fTop := low[i], fBot := high[i+2], fTime := time[i+2]
                fFVG := true
                break
        for i = 1 to lookback
            if close[i] < open[i] and time[i] <= lastSwingLowTime
                oTop := high[i], oBot := low[i], oTime := time[i]
                fOB := true
                break
        if fFVG and fOB
            activeDir := 1

    [activeDir, oTop, oBot, oTime, fTop, fBot, fTime, chochLvl, chochT]

core_st_logic() =>
    int tf_seconds = timeframe.in_seconds(timeframe.period)
    string active_ma = in_st_ma
    int active_period = in_st_period
    float active_mult = in_st_mult
    int active_cb = in_st_cb
    
    if in_st_opMode == MODE_AUTO
        if tf_seconds < 86400
            active_ma := MA_VAWMA, active_period := 50, active_mult := 3.0, active_cb := 2
        else if tf_seconds >= 86400 and tf_seconds < 604800
            active_ma := MA_VAWMA, active_period := 100, active_mult := 4.0, active_cb := 2
        else
            active_ma := MA_VAWMA, active_period := 120, active_mult := 5.0, active_cb := 2

    float vol = nz(volume, 1)
    float tr_val = ta.tr(true)
    float atr_val = na
    switch active_ma
        MA_SMA   => atr_val := ta.sma(tr_val, active_period)
        MA_EMA   => atr_val := ta.ema(tr_val, active_period)
        MA_WMA   => atr_val := ta.wma(tr_val, active_period)
        MA_VWMA  => atr_val := ta.vwma(tr_val, active_period)
        MA_VAWMA => atr_val := ta.wma(tr_val * vol, active_period) / ta.wma(vol, active_period)
        => atr_val := ta.sma(tr_val, active_period)
    
    float hl2 = math.avg(high, low)
    float basic_up = hl2 + (active_mult * atr_val)
    float basic_dn = hl2 - (active_mult * atr_val)

    float upper = na, float lower = na
    int trend = na, int bars_above = na, int bars_below = na
    
    upper := nz(upper[1], basic_up)
    lower := nz(lower[1], basic_dn)
    trend := nz(trend[1], 1)
    bars_above := nz(bars_above[1], 0)
    bars_below := nz(bars_below[1], 0)
    
    if trend == -1
        if close > upper
            bars_above += 1
        else
            bars_above := 0 
    else if trend == 1
        if close < lower
            bars_below += 1
        else
            bars_below := 0
            
    bool flip_to_bull = trend == -1 and bars_above >= active_cb
    bool flip_to_bear = trend == 1 and bars_below >= active_cb
    
    if flip_to_bull
        trend := 1, bars_above := 0, lower := basic_dn 
    else if flip_to_bear
        trend := -1, bars_below := 0, upper := basic_up 
        
    if trend == 1 and not flip_to_bull
        lower := basic_dn > lower ? basic_dn : lower
    else if trend == -1 and not flip_to_bear
        upper := basic_up < upper ? basic_up : upper
        
    [upper, lower, trend]
