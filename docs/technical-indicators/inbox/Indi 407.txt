f_line_price(
     int x1,
     float y1,
     int x2,
     float y2,
     int targetX
) =>
    float result = na

    if (
         not na(x1) and
         not na(x2) and
         x1 != x2
    )
        float slope = (y2 - y1) / float(x2 - x1)
        result := y1 + slope * float(targetX - x1)

    result

// ============================================================================
// JOINT STRUCTURE SCORE
// ============================================================================

f_score_structure(
     int ux1,
     float uy1,
     int ux2,
     float uy2,
     int lx1,
     float ly1,
     int lx2,
     float ly2
) =>
    float finalScore = na

    int upperTouchesLocal = 0
    int lowerTouchesLocal = 0
    int upperViolationsLocal = 0
    int lowerViolationsLocal = 0

    int lastUpperTouchX = na
    int lastLowerTouchX = na

    bool candidateValid = (
         ux2 > ux1 and
         lx2 > lx1 and
         ux2 - ux1 >= minAnchorSpacing and
         lx2 - lx1 >= minAnchorSpacing
    )

    if candidateValid

        float uSlope = (uy2 - uy1) / float(ux2 - ux1)
        float lSlope = (ly2 - ly1) / float(lx2 - lx1)

        int structureStart = math.max(
             ux1,
             lx1
        )

        float upperAtStart = f_line_price(
             ux1,
             uy1,
             ux2,
             uy2,
             structureStart
        )

        float lowerAtStart = f_line_price(
             lx1,
             ly1,
             lx2,
             ly2,
             structureStart
        )

        float widthAtStart = (
             upperAtStart -
             lowerAtStart
        )

        float upperCurrent = f_line_price(
             ux1,
             uy1,
             ux2,
             uy2,
             bar_index
        )

        float lowerCurrent = f_line_price(
             lx1,
             ly1,
             lx2,
             ly2,
             bar_index
        )

        float currentWidthLocal = (
             upperCurrent -
             lowerCurrent
        )

        bool convergingCandidate = (
             uSlope < lSlope
        )

        float candidateApexX = na

        float denominator = (
             uSlope -
             lSlope
        )

        if math.abs(denominator) > 0.0000001

            float upperIntercept = (
                 uy1 -
                 uSlope *
                 float(ux1)
            )

            float lowerIntercept = (
                 ly1 -
                 lSlope *
                 float(lx1)
            )

            candidateApexX := (
                 lowerIntercept -
                 upperIntercept
            ) / denominator

        bool apexAlreadyPassed = (
             convergingCandidate and
             not na(candidateApexX) and
             candidateApexX <= float(bar_index)
        )

        bool orderedAtStart = (
             widthAtStart > 0.0
        )

        bool orderedNow = (
             currentWidthLocal > 0.0
        )

        if (
             orderedAtStart and
             orderedNow and
             not apexAlreadyPassed
        )

            int availableBars = (
                 bar_index -
                 structureStart
            )

            int scanBars = math.min(
                 autoLookbackBars,
                 math.max(
                      0,
                      availableBars
                 )
            )

            for offset = 0 to scanBars

                int testX = (
                     bar_index -
                     offset
                )

                if testX >= structureStart

                    float testUpper = f_line_price(
                         ux1,
                         uy1,
                         ux2,
                         uy2,
                         testX
                    )

                    float testLower = f_line_price(
                         lx1,
                         ly1,
                         lx2,
                         ly2,
                         testX
                    )

                    float testATR = atr[offset]

                    float touchTol = (
                         not na(testATR)
                         ? testATR * touchToleranceATR
                         : 0.0
                    )

                    float violationTol = (
                         not na(testATR)
                         ? testATR * violationToleranceATR
                         : 0.0
                    )

                    bool upperNear = (
                         math.abs(
                              high[offset] -
                              testUpper
                         ) <= touchTol
                    )

                    bool lowerNear = (
                         math.abs(
                              low[offset] -
                              testLower
                         ) <= touchTol
                    )

                    bool upperViolation = (
                         close[offset] >
                         testUpper +
                         violationTol
                    )

                    bool lowerViolation = (
                         close[offset] <
                         testLower -
                         violationTol
                    )

                    if upperNear

                        bool spacedUpper = (
                             na(lastUpperTouchX) or
                             math.abs(
                                  testX -
                                  lastUpperTouchX
                             ) >= minTouchSpacing
                        )

                        if spacedUpper
                            upperTouchesLocal += 1
                            lastUpperTouchX := testX

                    if lowerNear

                        bool spacedLower = (
                             na(lastLowerTouchX) or
                             math.abs(
                                  testX -
                                  lastLowerTouchX
                             ) >= minTouchSpacing
                        )

                        if spacedLower
                            lowerTouchesLocal += 1
                            lastLowerTouchX := testX

                    if upperViolation
                        upperViolationsLocal += 1

                    if lowerViolation
                        lowerViolationsLocal += 1

            int totalTouchesLocal = (
                 upperTouchesLocal +
                 lowerTouchesLocal
            )

            int totalViolationsLocal = (
                 upperViolationsLocal +
                 lowerViolationsLocal
            )

            int minTouches = math.min(
                 upperTouchesLocal,
                 lowerTouchesLocal
            )

            int maxTouches = math.max(
                 upperTouchesLocal,
                 lowerTouchesLocal
            )

            float touchScore = math.min(
                 35.0,
                 float(totalTouchesLocal) *
                 5.0
            )

            float balanceFactor = (
                 totalTouchesLocal > 0
                 ? (
                      1.0 -
                      math.abs(
                           float(
                                upperTouchesLocal -
                                lowerTouchesLocal
                           )
                      ) /
                      float(totalTouchesLocal)
                   )
                 : 0.0
            )

            float balanceScore = (
                 balanceFactor *
                 15.0
            )

            float violationScore = math.max(
                 0.0,
                 25.0 -
                 float(totalViolationsLocal) *
                 3.0
            )

            int upperSpan = (
                 ux2 -
                 ux1
            )

            int lowerSpan = (
                 lx2 -
                 lx1
            )

            float averageSpan = (
                 float(upperSpan) +
                 float(lowerSpan)
            ) / 2.0

            float spanScore = math.min(
                 10.0,
                 averageSpan /
                 15.0
            )

            int newestAnchor = math.max(
                 ux2,
                 lx2
            )

            float recencyScore = math.max(
                 0.0,
                 10.0 -
                 float(
                      bar_index -
                      newestAnchor
                 ) *
                 0.15
            )

            float candidateWidthATR = (
                 atr > 0.0
                 ? currentWidthLocal /
                   atr
                 : na
            )

            float widthScore = 0.0

            if not na(candidateWidthATR)

                if (
                     candidateWidthATR >= minStructureWidthATR and
                     candidateWidthATR <= maxStructureWidthATR
                )

                    widthScore := 5.0

                else if (
                     candidateWidthATR <
                     minStructureWidthATR
                )

                    widthScore := math.max(
                         0.0,
                         (
                              candidateWidthATR /
                              minStructureWidthATR
                         ) *
                         5.0
                    )

                else

                    widthScore := math.max(
                         0.0,
                         5.0 -
                         (
                              candidateWidthATR -
                              maxStructureWidthATR
                         ) *
                         0.25
                    )

            finalScore := (
                 touchScore +
                 balanceScore +
                 violationScore +
                 spanScore +
                 recencyScore +
                 widthScore
            )

            // -----------------------------------------------------------
            // Bilateral validation
            // -----------------------------------------------------------

            if minTouches < 2

                finalScore := math.min(
                     finalScore,
                     59.0
                )

            else if minTouches < 3

                finalScore := math.min(
                     finalScore,
                     84.0
                )

            // -----------------------------------------------------------
            // Stronger imbalance penalty
            // -----------------------------------------------------------

            if minTouches > 0

                float touchRatio = float(maxTouches) / float(minTouches)

                if touchRatio > balancePenaltyStart

                    float imbalancePenalty = (touchRatio - balancePenaltyStart) * balancePenaltyStrength

                    finalScore -= imbalancePenalty

                // 2:1 or worse can never be Excellent.
                if touchRatio >= 2.0
                    finalScore := math.min(finalScore, 84.0)

                // 2.5:1 or worse gets capped below strong Good.
                if touchRatio >= 2.5
                    finalScore := math.min(finalScore, 74.0)

                // 3:1 or worse is only Fair.
                if touchRatio >= 3.0
                    finalScore := math.min(finalScore, 59.0)

            finalScore := math.max(0.0, math.min(100.0, finalScore))

     upperX1 != upperX2
)

bool validLower = (
     not na(lowerX1) and
     not na(lowerX2) and
     not na(lowerY1) and
     not na(lowerY2) and
     lowerX1 != lowerX2
)

bool structureValid = (
     validUpper and
     validLower
)

// ============================================================================
// CURRENT VALUES
// ============================================================================

float upperNow = (
     validUpper
     ? f_line_price(
          upperX1,
          upperY1,
          upperX2,
          upperY2,
          bar_index
       )
     : na
)

float lowerNow = (
     validLower
     ? f_line_price(
          lowerX1,
          lowerY1,
          lowerX2,
          lowerY2,
          bar_index
       )
     : na
)

float structureWidth = (
     structureValid
     ? upperNow -
       lowerNow
     : na
)

bool boundariesOrdered = (
     structureValid and
     not na(structureWidth) and
     structureWidth > 0.0
)

// ============================================================================
// SLOPES / GEOMETRY
// ============================================================================

float upperSlope = (
     validUpper
     ? (
          upperY2 -
          upperY1
       ) /
       float(
          upperX2 -
          upperX1
       )
     : na
)

float lowerSlope = (
     validLower
     ? (
          lowerY2 -
          lowerY1
       ) /
       float(
          lowerX2 -
          lowerX1
       )
     : na
)

bool converging = (
     structureValid and
     upperSlope <
     lowerSlope
)

bool expanding = (
     structureValid and
     upperSlope >
     lowerSlope
)

float slopeDiff = (
     structureValid
     ? math.abs(
          upperSlope -
          lowerSlope
       )
     : na
)

float largestSlope = (
     structureValid
     ? math.max(
          math.abs(
               upperSlope
          ),
          math.abs(
               lowerSlope
          )
       )
     : na
)

bool roughlyParallel = (
     structureValid and
     (
          largestSlope == 0.0 or
          slopeDiff <=
          largestSlope *
          0.10
     )
)

// ============================================================================
// APEX
// ============================================================================

float apexX = na

if (
     structureValid and
     not roughlyParallel
)

    float upperIntercept = (
         upperY1 -
         upperSlope *
         float(
              upperX1
         )
    )

    float lowerIntercept = (
         lowerY1 -
         lowerSlope *
         float(
              lowerX1
         )
    )

    float denominator = (
         upperSlope -
         lowerSlope
    )

    if (
         math.abs(
              denominator
         ) >
         0.0000001
    )

        apexX := (lowerIntercept - upperIntercept) / denominator

float barsToApex = (
     converging and
     not na(apexX)
     ? apexX -
       float(
            bar_index
       )
     : na
)

bool apexPassed = (
     converging and
     not na(
          barsToApex
     ) and
     barsToApex <= 0.0
)

// ============================================================================
// WIDTH
// ============================================================================

float widthATR = (
     boundariesOrdered and
     atr > 0.0
     ? structureWidth /
       atr
     : na
)

// ============================================================================
// CONVERGENCE STRENGTH
// ============================================================================

float projectedUpper = (
     structureValid
     ? f_line_price(
          upperX1,
          upperY1,
          upperX2,
          upperY2,
          bar_index +
          convergenceProjectionBars
       )
     : na
)

float projectedLower = (
     structureValid
     ? f_line_price(
          lowerX1,
          lowerY1,
          lowerX2,
          lowerY2,
          bar_index +
          convergenceProjectionBars
       )
     : na
)

float projectedWidth = (
     structureValid
     ? projectedUpper -
       projectedLower
     : na
)

float convergencePct = (
     converging and
     not na(
          structureWidth
     ) and
     structureWidth > 0.0 and
     not na(
          projectedWidth
     )
     ? (
          (
               structureWidth -
               projectedWidth
          ) /
          structureWidth
       ) *
       100.0
     : na
)

string convergenceText = (
     "N/A"
)

if (
     converging and
     not na(
          convergencePct
     )
)

    if (
         convergencePct >=
         strongConvergencePct
    )

        convergenceText := "Strong"

    else if (
         convergencePct >=
         weakConvergencePct
    )

        convergenceText := "Moderate"

    else

        convergenceText := "Weak"

     converging and
     not na(
          widthATR
     ) and
     widthATR <=
     lateStageWidthATR

bool lateByApex = (
     converging and
     not na(
          barsToApex
     ) and
     barsToApex > 0.0 and
     barsToApex <=
     lateStageApexBars
)

          lateByWidth or
          lateByApex
     )
)

     "Unavailable"

            )

             upperSlope <= 0.0 and
             lowerSlope > 0.0
        )

            structureType := (
                 "Ascending Triangle/Wedge"
            )

        else

            structureType := (
                 "Converging Wedge"
            )

    else if expanding

        )

          lowerX2[1]
     )

int liveLowerTouches = 0

int liveUpperViolations = 0
int liveLowerViolations = 0

int lastUpperTouchBar = na
int lastLowerTouchBar = na

if structureValid

    int metricStartX = math.max(
         upperX1,
         lowerX1
    )

    int metricAvailableBars = (
         bar_index -
         metricStartX
    )

    int metricScanBars = math.min(
         autoLookbackBars,
         math.max(
              0,
              metricAvailableBars
         )
    )

    for offset = 0 to metricScanBars

        int testX = (
             bar_index -
             offset
        )

        if (
             testX >=
             metricStartX
        )

            float metricUpper = f_line_price(
                 upperX1,
                 upperY1,
                 upperX2,
                 upperY2,
                 testX
            )

            float metricLower = f_line_price(
                 lowerX1,
                 lowerY1,
                 lowerX2,
                 lowerY2,
                 testX
            )

            float metricATR = (
                 atr[offset]
            )

            float metricTouchTol = (
                 not na(metricATR)
                 ? metricATR *
                   touchToleranceATR
                 : 0.0
            )

            float metricViolationTol = (
                 not na(metricATR)
                 ? metricATR *
                   violationToleranceATR
                 : 0.0
            )

            bool metricUpperNear = (
                 math.abs(
                      high[offset] -
                      metricUpper
                 ) <=
                 metricTouchTol
            )

            bool metricLowerNear = (
                 math.abs(
                      low[offset] -
                      metricLower
                 ) <=
                 metricTouchTol
            )

            bool metricUpperViolation = (
                 close[offset] >
                 metricUpper +
                 metricViolationTol
            )

            bool metricLowerViolation = (
                 close[offset] <
                 metricLower -
                 metricViolationTol
            )

            if metricUpperNear

                bool spacedUpper = (
                     na(
                          lastUpperTouchBar
                     ) or
                     math.abs(
                          testX -
                          lastUpperTouchBar
                     ) >=
                     minTouchSpacing
                )

                if spacedUpper

                    liveUpperTouches += 1
                    lastUpperTouchBar := testX

            if metricLowerNear

                bool spacedLower = (
                     na(
                          lastLowerTouchBar
                     ) or
                     math.abs(
                          testX -
                          lastLowerTouchBar
                     ) >=
                     minTouchSpacing
                )

                if spacedLower

                    liveLowerTouches += 1
                    lastLowerTouchBar := testX

            if metricUpperViolation

                liveUpperViolations += 1

            if metricLowerViolation

                liveLowerViolations += 1

     structureValid
     ? f_score_structure(

          upperY1,
          upperX2,
          upperY2,
          lowerX1,
          lowerY1,
          lowerX2,
          lowerY2
       )
     : na

    breakBar := na

             bar_index -

     belowBreakCount ==
     breakoutConfirmBars

     breakDirection == 0 and

    breakDirection := 1
    breakBar := bar_index

    frozenQualityScore := liveQualityScore

    frozenUpperTouches := liveUpperTouches
    frozenLowerTouches := liveLowerTouches

    frozenUpperViolations := liveUpperViolations
    frozenLowerViolations := liveLowerViolations

if (
     breakDirection == 0 and
     confirmedBreakDown
)

    breakDirection := -1
    breakBar := bar_index

    frozenQualityScore := liveQualityScore

    frozenUpperTouches := liveUpperTouches
    frozenLowerTouches := liveLowerTouches

    frozenUpperViolations := liveUpperViolations
    frozenLowerViolations := liveLowerViolations

if (
     apexPassed and
     not frozenAtApex and
     na(
          frozenQualityScore

              liveQualityScore

         : autoSelectionScore

    frozenUpperViolations := liveUpperViolations
    frozenLowerViolations := liveLowerViolations

    frozenAtApex := true
