// ══════════════════════════════════════════════════════════════
// ZigZag pivot store
// ══════════════════════════════════════════════════════════════
var array<float> pvP = array.new_float()
var array<int>   pvB = array.new_int()
var array<int>   pvT = array.new_int()   // 1 high, -1 low

pushPivot(float price, int bar, int typ) =>
    bool changed = false
    n = array.size(pvT)
    if n == 0
        array.push(pvP, price)
        array.push(pvB, bar)
        array.push(pvT, typ)
        changed := true
    else if array.get(pvT, n - 1) == typ
        prev = array.get(pvP, n - 1)
        if (typ == 1 and price > prev) or (typ == -1 and price < prev)
            array.set(pvP, n - 1, price)
            array.set(pvB, n - 1, bar)
            changed := true
    else
        array.push(pvP, price)
        array.push(pvB, bar)
        array.push(pvT, typ)
        changed := true
    if array.size(pvT) > 32
        array.shift(pvP)
        array.shift(pvB)
        array.shift(pvT)
    changed

ph = ta.pivothigh(high, depth, depth)
pl = ta.pivotlow(low, depth, depth)
bool newPivot = false
if not na(ph)
    if pushPivot(ph, bar_index - depth, 1)
        newPivot := true
if not na(pl)
    if pushPivot(pl, bar_index - depth, -1)
        newPivot := true

// getters
gp(int i) => array.get(pvP, i)
gb(int i) => array.get(pvB, i)
gt_(int i) => array.get(pvT, i)

// ══════════════════════════════════════════════════════════════
// Validation functions
// ══════════════════════════════════════════════════════════════
upAt(int s) => gt_(s) == -1

impValid(int s) =>
    up = gt_(s) == -1
    p0 = gp(s)
    p1 = gp(s + 1)
    p2 = gp(s + 2)
    p3 = gp(s + 3)
    p4 = gp(s + 4)
    p5 = gp(s + 5)
    w1 = math.abs(p1 - p0)
    w3 = math.abs(p3 - p2)
    w5 = math.abs(p5 - p4)
    dir = up ? (p1 > p0 and p3 > p1 and p5 > p3 and p2 < p1 and p4 < p3) : (p1 < p0 and p3 < p1 and p5 < p3 and p2 > p1 and p4 > p3)
    rr1 = up ? p2 > p0 : p2 < p0
    rr2 = not (w3 < w1 and w3 < w5)
    rr3 = up ? p4 > p1 : p4 < p1
    dir and rr1 and rr2 and rr3

abcValid(int s, bool up) =>
    p5 = gp(s)
    pA = gp(s + 1)
    pB = gp(s + 2)
    pC = gp(s + 3)
    up ? (pA < p5 and pB > pA and pC < pB and pB < p5) : (pA > p5 and pB < pA and pC > pB and pB > p5)

cl(float actual, float ideal, float tol) => math.max(0.0, 1.0 - math.abs(actual - ideal) / tol)

var array<label> poolT = array.new_label()
var array<box>   poolB = array.new_box()

clearAll() =>
    if array.size(poolL) > 0
        for i = 0 to array.size(poolL) - 1
            line.delete(array.get(poolL, i))
        array.clear(poolL)
    if array.size(poolT) > 0
        for i = 0 to array.size(poolT) - 1
            label.delete(array.get(poolT, i))
        array.clear(poolT)
    if array.size(poolB) > 0
        for i = 0 to array.size(poolB) - 1
            box.delete(array.get(poolB, i))
        array.clear(poolB)

addLbl(int x, float y, string t, string style, color c, color tc, string sz) =>
    array.push(poolT, label.new(x, y, t, style = style, color = c, textcolor = tc, size = sz))

// Analysis state (for the panel + alerts)
// ══════════════════════════════════════════════════════════════
var string st_count = "—"
var bool   st_up = true
var float  st_w2 = na
var float  st_w3 = na
var float  st_w4 = na
var float  st_w5 = na
var float  st_qual = na
var float  st_q5 = na
var bool   st_alt = false
var bool   st_valid = false
var bool   st_full = false
var bool   st_current = false
var bool   st_w5ext = false
var float  st_inval = na
var int    st_startBar = -1

// capture pre-recount state for transition detection
bool wasValid = st_valid
bool wasFull = st_full
int  wasStart = st_startBar
bool evFound = false
bool evLost = false
bool evFull = false

// ══════════════════════════════════════════════════════════════
// Relabel on each new pivot
// ══════════════════════════════════════════════════════════════
if newPivot
    clearAll()
    n = array.size(pvT)

    // ZigZag
    if showZZ and n >= 2
        for i = 1 to n - 1
            array.push(poolL, line.new(gb(i - 1), gp(i - 1), gb(i), gp(i), color = zzCol, width = 1))

    // Sliding-window search: most recent end pivot wins; full count preferred at same end
    int bi = -1
    bool full = false
    int endIdx = -1
    if n >= 6
        for e = n - 1 to 5
            int s9 = e - 8
            int s6 = e - 5
            if s9 >= 0 and impValid(s9) and abcValid(s9 + 5, upAt(s9))
                bi := s9
                full := true
                endIdx := e
                break
            if impValid(s6)
                bi := s6
                full := false
                endIdx := e
                break

    st_valid := bi != -1
    st_full := full

    if bi != -1
        up = upAt(bi)
        st_up := up
        st_current := endIdx == n - 1
        st_count := full ? "1-5 + A-B-C" : "1-5 impulse"
        st_startBar := gb(bi)
        wc = up ? upCol : dnCol

        p0 = gp(bi)
        p1 = gp(bi + 1)
        p2 = gp(bi + 2)
        p3 = gp(bi + 3)
        p4 = gp(bi + 4)
        p5 = gp(bi + 5)
        int b1 = gb(bi + 1)
        int b2 = gb(bi + 2)
        int b3 = gb(bi + 3)
        int b4 = gb(bi + 4)
        int b5 = gb(bi + 5)

        w1 = math.abs(p1 - p0)
        w3abs = math.abs(p3 - p2)
        // Fib ratios
        st_w2 := w1 != 0 ? math.abs(p2 - p1) / w1 : na
        st_w3 := w1 != 0 ? w3abs / w1 : na
        st_w4 := w3abs != 0 ? math.abs(p4 - p3) / w3abs : na
        st_w5 := w1 != 0 ? math.abs(p5 - p4) / w1 : na
        // Alternation: wave 2 and wave 4 retraces differ meaningfully
        w4rOfW1 = w1 != 0 ? math.abs(p4 - p3) / w1 : na
        st_alt := not na(st_w2) and not na(w4rOfW1) and math.abs(st_w2 - w4rOfW1) > 0.2
        // Quality (textbook fit); W5 scored vs best of ~1.0×W1 and extended 0.618×(W1+W3)
        q2 = cl(st_w2, 0.559, 0.3)
        q3 = st_w3 >= 1.618 ? 1.0 : math.max(0.0, st_w3 / 1.618)
        q4 = cl(st_w4, 0.382, 0.3)
        extIdeal = w1 != 0 ? 0.618 * (w1 + w3abs) / w1 : na
        q5a = cl(st_w5, 1.0, 0.5)
        q5b = na(extIdeal) ? 0.0 : cl(st_w5, extIdeal, 0.5)
        st_w5ext := q5b > q5a
        st_q5 := math.max(q5a, q5b)
        st_qual := (q2 + q3 + q4 + st_q5) / 4.0 * 100.0
        // Invalidation: correction may not retrace 100% of the impulse
        st_inval := p0

        // Wave labels 1-5
        if showWaves
            addLbl(b1, p1, "1", up ? label.style_label_down : label.style_label_up, color.new(wc, 15), color.white, lSize)
            addLbl(b2, p2, "2", up ? label.style_label_up : label.style_label_down, color.new(wc, 15), color.white, lSize)
            addLbl(b3, p3, "3", up ? label.style_label_down : label.style_label_up, color.new(wc, 15), color.white, lSize)
            addLbl(b4, p4, "4", up ? label.style_label_up : label.style_label_down, color.new(wc, 15), color.white, lSize)
            addLbl(b5, p5, "5", up ? label.style_label_down : label.style_label_up, color.new(wc, 15), color.white, lSize)

        // Ratio annotations
        if showRatios
            addLbl(b2, p2, str.tostring(st_w2, "0.00"), label.style_none, color.new(color.black, 100), color.new(wc, 0), size.tiny)
            addLbl(b3, p3, str.tostring(st_w3, "0.00"), label.style_none, color.new(color.black, 100), color.new(wc, 0), size.tiny)
            addLbl(b4, p4, str.tostring(st_w4, "0.00"), label.style_none, color.new(color.black, 100), color.new(wc, 0), size.tiny)
            addLbl(b5, p5, str.tostring(st_w5, "0.00"), label.style_none, color.new(color.black, 100), color.new(wc, 0), size.tiny)

        // A-B-C
        if full and showABC
            pA = gp(bi + 6)
            pB = gp(bi + 7)
            pC = gp(bi + 8)
            int bA = gb(bi + 6)
            int bB = gb(bi + 7)
            int bC = gb(bi + 8)
            addLbl(bA, pA, "A", up ? label.style_label_up : label.style_label_down, color.new(corCol, 15), color.white, lSize)
            addLbl(bB, pB, "B", up ? label.style_label_down : label.style_label_up, color.new(corCol, 15), color.white, lSize)
            addLbl(bC, pC, "C", up ? label.style_label_up : label.style_label_down, color.new(corCol, 15), color.white, lSize)

        // Fib targets for wave 3 / 5 (guide)
        if showFib
            t3 = up ? p2 + w1 * w3ext : p2 - w1 * w3ext
            t5 = up ? p4 + w1 * w5ext : p4 - w1 * w5ext
            int projX = b5 + depth * 3
            array.push(poolL, line.new(b2, t3, projX, t3, color = fibCol, style = line.style_dotted, width = 1))
            array.push(poolL, line.new(b4, t5, projX, t5, color = fibCol, style = line.style_dashed, width = 1))
            addLbl(projX, t3, "W3 " + str.tostring(w3ext) + " " + str.tostring(t3, format.mintick), label.style_label_left, color.new(fibCol, 80), fibCol, size.small)
            addLbl(projX, t5, "W5 " + str.tostring(w5ext) + " " + str.tostring(t5, format.mintick), label.style_label_left, color.new(fibCol, 80), fibCol, size.small)

        // Elliott base channel: line through 2 & 4, parallel through 3
        if showChan and b4 != b2
            slope = (p4 - p2) / (b4 - b2)
            int projX2 = b5 + depth * 3
            yEnd24 = p2 + slope * (projX2 - b2)
            array.push(poolL, line.new(b2, p2, projX2, yEnd24, color = chanCol, width = 1))
            offset = p3 - (p2 + slope * (b3 - b2))
            yStartPar = p2 + offset
            yEndPar = yEnd24 + offset
            array.push(poolL, line.new(b2, yStartPar, projX2, yEndPar, color = chanCol, style = line.style_dashed, width = 1))

        // Next-wave target zone (correction retrace of the whole 1-5)
        if showZone and not full
            float moveLen = p5 - p0
            float z1 = p5 - moveLen * 0.382
            float z2 = p5 - moveLen * 0.618
            float ztop = math.max(z1, z2)
            float zbot = math.min(z1, z2)
            array.push(poolB, box.new(b5, ztop, b5 + depth * 4, zbot, border_color = na, bgcolor = zoneCol, text = "Wave 2/4-style retrace zone", text_size = size.tiny, text_color = color.new(#ab47bc, 0)))

        // Invalidation level
        if showInval
            int projX3 = b5 + depth * 3
            array.push(poolL, line.new(gb(bi), p0, projX3, p0, color = invalCol, width = 2))
            addLbl(projX3, p0, (up ? "Invalid below " : "Invalid above ") + str.tostring(p0, format.mintick), label.style_label_left, color.new(invalCol, 80), invalCol, size.small)
    else
        // v2 FIX: reset state so the panel never shows stale metrics
        st_count := "—"
        st_w2 := na
        st_w3 := na
        st_w4 := na
        st_w5 := na
        st_qual := na
        st_q5 := na
        st_alt := false
        st_current := false
        st_w5ext := false
        st_inval := na
        st_startBar := -1

    // Transition events (extension of the same count does NOT re-fire)
    evFound := st_valid and (not wasValid or st_startBar != wasStart)
    evLost := wasValid and not st_valid
    evFull := st_valid and st_full and not wasFull

// ══════════════════════════════════════════════════════════════
// Data-window / strategy outputs
// ══════════════════════════════════════════════════════════════

plot(st_valid ? (st_up ? 1 : -1) : 0, "Count Direction", display = display.data_window, editable = false)
plot(st_valid ? (st_full ? 2 : 1) : 0, "Count Stage", display = display.data_window, editable = false)
plot(st_valid ? st_inval : na, "Invalidation Level", display = display.data_window, editable = false)
